## 📋 需求分析

### 核心功能
1. ✅ 编辑直播封面（cover_url）
2. ✅ 编辑标题（title）
3. ✅ 编辑简介（summary/description）
4. ✅ 编辑开播时间（start_time）
5. ✅ 更换关联专家（featured_expert_id）
6. ✅ 更换关联品牌（brand相关）
7. ✅ 编辑回放URL（replay_video_url）

### 触发入口
- **列表页**：长按或点击"⋮"按钮 → 显示【编辑】【删除】【详情】
- **详情页**：顶部"编辑"按钮 → 跳转到编辑页面

---

## 🎯 数据层级设计

根据后端设计文档，直播数据分为两层：

### Room 层级（live_rooms表）
```typescript
{
  id: string              // 直播间ID
  title: string           // 直播间标题
  description: string     // 详细描述
  summary: string         // 简介摘要（最多255字符）
  cover_url: string       // 封面URL
  category_id: string     // 分类ID
  owner_id: string        // 创建者ID
}
```

### Session 层级（live_sessions表）
```typescript
{
  id: string                    // 场次ID
  room_id: string               // 所属直播间
  title: string                 // 场次标题（可选，覆盖room标题）
  summary: string               // 场次摘要
  start_time: string            // 开播时间
  featured_expert_id: string    // 特邀专家ID
  replay_video_url: string      // 回放URL
  status: 'pending' | 'live' | 'ended'
}
```

  // uni-datetime-picker 会自动显示

## 🔧 Store层实现（myLive.ts）

```typescript
// src/store/myLive.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import { request } from '@/utils/request'
import type { LiveRoom, LiveSession, Category, Expert } from '@/types'

export const useMyLiveStore = defineStore('myLive', () => {
  const myRooms = ref<LiveRoom[]>([])
  const loading = ref(false)
  
  /**
   * 获取我的直播间列表
   */
  async function loadMyRooms() {
    loading.value = true
    try {
      const res = await request({
        url: '/rooms',
        method: 'GET',
        data: {
          owner_id: 'me' // 后端根据JWT Token筛选
        }
      })
      myRooms.value = res.data.items
    } finally {
      loading.value = false
    }
  }
  
  /**
   * 获取直播间详情
   */
  async function getRoomDetail(roomId: string) {
    const res = await request({
      url: `/rooms/${roomId}`,
      method: 'GET'
    })
    return res.data
  }
  
  /**
   * 更新直播间信息
   */
  async function updateRoom(roomId: string, data: Partial<LiveRoom>) {
    const res = await request({
      url: `/rooms/${roomId}`,
      method: 'PATCH',
      data
    })
    
    // 乐观更新本地列表
    const index = myRooms.value.findIndex(r => r.id === roomId)
    if (index !== -1) {
      myRooms.value[index] = { ...myRooms.value[index], ...data }
    }
    
    return res.data
  }
  
  /**
   * 更新场次信息
   */
  async function updateSession(sessionId: string, data: Partial<LiveSession>) {
    const res = await request({
      url: `/sessions/${sessionId}`,
      method: 'PATCH',
      data
    })
    return res.data
  }
  
  /**
   * 删除直播间
   */
  async function deleteRoom(roomId: string) {
    await request({
      url: `/rooms/${roomId}`,
      method: 'DELETE'
    })
    
    // 从本地列表移除
    myRooms.value = myRooms.value.filter(r => r.id !== roomId)
  }
  
  /**
   * 获取分类列表
   */
  async function getCategories(): Promise<Category[]> {
    const res = await request({
      url: '/categories',
      method: 'GET'
    })
    return res.data.items
  }
  
  /**
   * 获取专家列表
   */
  async function getExperts(): Promise<Expert[]> {
    const res = await request({
      url: '/experts',
      method: 'GET'
    })
    return res.data.items
  }
  
  return {
    myRooms,
    loading,
    loadMyRooms,
    getRoomDetail,
    updateRoom,
    updateSession,
    deleteRoom,
    getCategories,
    getExperts
  }
})
```

### 文件上传工具（utils/upload.ts）

```typescript
/**
 * 上传图片
 */
export async function uploadImage(): Promise<string> {
  return new Promise((resolve, reject) => {
    uni.chooseImage({
      count: 1,
      sizeType: ['compressed'],
      sourceType: ['album', 'camera'],
      success: (res) => {
        const filePath = res.tempFilePaths[0]
        
        uni.uploadFile({
          url: 'https://mp.dayilive.com/api/core/upload/image',
          filePath,
          name: 'file',
          header: {
            'Authorization': `Bearer ${getToken()}`
          },
          success: (uploadRes) => {
            const data = JSON.parse(uploadRes.data)
            if (data.code === 200) {
              resolve(data.data.url)
            } else {
              reject(new Error(data.message))
            }
          },
          fail: reject
        })
      },
      fail: reject
    })
  })
}
```
