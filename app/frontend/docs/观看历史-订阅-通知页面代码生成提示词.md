```typescript
/**
 * 安全的图片URL获取（复用已有的URL校验逻辑）
 * @param url 原始URL
 * @param type 图片类型
 * @returns 安全的图片URL或占位图
 * @example getImageUrl('https://cdn.dayilive.com/cover.jpg', 'cover')
 */
export function getImageUrl(url: string | undefined | null, type: 'cover' | 'avatar' = 'cover'): string {
  if (!url || typeof url !== 'string' || url.trim() === '') {
    return type === 'cover' ? DEFAULT_COVER_PLACEHOLDER : DEFAULT_AVATAR_PLACEHOLDER
  }
  
  const trimmedUrl = url.trim()
  
  // 特殊处理：禁止SVG data URI（可能包含script标签）
  if (trimmedUrl.toLowerCase().startsWith('data:image/svg')) {
    console.warn('[Security] SVG data URI blocked')
    return type === 'cover' ? DEFAULT_COVER_PLACEHOLDER : DEFAULT_AVATAR_PLACEHOLDER
  }
  
  // 复用已有的URL安全校验
  if (!validateUrl(trimmedUrl)) {
    console.warn('[Security] Blocked unsafe image URL')
    return type === 'cover' ? DEFAULT_COVER_PLACEHOLDER : DEFAULT_AVATAR_PLACEHOLDER
  }
  
  // 生产环境域名白名单校验（图片专用）
  if (import.meta.env.PROD && trimmedUrl.startsWith('http')) {
    try {
      const urlObj = new URL(trimmedUrl)
      const isAllowed = TRUSTED_IMAGE_DOMAINS.some(domain => {
        return urlObj.hostname === domain || urlObj.hostname.endsWith('.' + domain)
      })
      
      if (!isAllowed) {
        console.warn('[Security] Image domain not in whitelist:', urlObj.hostname)
        return type === 'cover' ? DEFAULT_COVER_PLACEHOLDER : DEFAULT_AVATAR_PLACEHOLDER
      }
    } catch {
      return type === 'cover' ? DEFAULT_COVER_PLACEHOLDER : DEFAULT_AVATAR_PLACEHOLDER
    }
  }
  
  // 相对路径转绝对路径
  if (trimmedUrl.startsWith('/')) {
    const baseUrl = import.meta.env.VITE_API_BASE_URL || 'https://mp.dayilive.com'
    return `${baseUrl}${trimmedUrl}`
  }
  
  return trimmedUrl
}
```

```typescript
// 分组算法
function groupHistoryByDate(items: WatchHistory[]): GroupedHistory[] {
  const today = dayjs().startOf('day')
  const yesterday = dayjs().subtract(1, 'day').startOf('day')
  
  const groups = {
    today: [],
    yesterday: [],
    earlier: []
  }
  
  items.forEach(item => {
    const watchedAt = dayjs(item.watched_at)
    if (watchedAt.isSame(today, 'day')) {
      groups.today.push(item)
    } else if (watchedAt.isSame(yesterday, 'day')) {
      groups.yesterday.push(item)
    } else {
      groups.earlier.push(item)
    }
  })
  
  return [
    { date: '今天', items: groups.today },
    { date: '昨天', items: groups.yesterday },
    { date: '更早', items: groups.earlier }
  ].filter(group => group.items.length > 0)
}
```

**逻辑实现**：
```typescript
const touchStartX = ref(0)
const touchStartY = ref(0)
const isSwiping = ref(false)

const handleTouchStart = (e: any, index: number) => {
  touchStartX.value = e.touches[0].clientX
  touchStartY.value = e.touches[0].clientY
  isSwiping.value = false
}

const handleTouchMove = (e: any, index: number) => {
  const deltaX = e.touches[0].clientX - touchStartX.value
  const deltaY = e.touches[0].clientY - touchStartY.value
  
  // 判断是否为横向滑动
  if (Math.abs(deltaX) > Math.abs(deltaY) && Math.abs(deltaX) > 10) {
    isSwiping.value = true
    
    // 阻止页面滚动（解决滚动冲突）
    e.preventDefault()
    e.stopPropagation()
    
    // 限制滑动范围：0 ~ -120rpx
    const translateX = Math.min(0, Math.max(-120, deltaX * 0.5))
    
    const items = groupedHistory.value.flatMap(g => g.items)
    
    // 关闭其他已打开的卡片（解决多卡片冲突）
    items.forEach((item, i) => {
      if (i !== index && item.translateX !== 0) {
        item.translateX = 0
      }
    })
    
    // 更新当前项的translateX
    items[index].translateX = translateX
  }
}

const handleTouchEnd = (index: number) => {
  const items = groupedHistory.value.flatMap(g => g.items)
  const currentTranslateX = items[index].translateX || 0
  
  // 滑动超过一半则展开删除按钮，否则复位
  if (currentTranslateX < -60) {
    items[index].translateX = -120
  } else {
    items[index].translateX = 0
  }
  
  isSwiping.value = false
}
```

```typescript
import dayjs from 'dayjs'

// 判断场次状态
const getSessionTabStatus = (session: any): 'pending' | 'notified' | 'expired' => {
  const now = dayjs()
  const startTime = dayjs(session.start_time)
  const endTime = dayjs(session.end_time)
  
  // 待开播：开始时间晚于当前时间
  if (startTime > now && session.status === 'scheduled') {
    return 'pending'
  }
  
  // 已通知：开始时间早于当前，结束时间晚于当前，状态为live
  if (startTime <= now && endTime >= now && session.status === 'live') {
    return 'notified'
  }
  
  // 已过期：结束时间早于当前，已结束
  if (endTime < now && session.status === 'ended') {
    return 'expired'
  }
  
  // 默认返回待开播
  return 'pending'
}

// 获取状态显示文本
const getStatusText = (session: any): string => {
  const status = getSessionTabStatus(session)
  const startTime = dayjs(session.start_time)
  const now = dayjs()
  
  switch (status) {
    case 'pending':
      const diffDays = startTime.diff(now, 'day')
      if (diffDays === 0) {
        return `今天 ${startTime.format('HH:mm')} 开播`
      } else if (diffDays === 1) {
        return `明天 ${startTime.format('HH:mm')} 开播`
      } else {
        return `${startTime.format('MM月DD日 HH:mm')} 开播`
      }
    case 'notified':
      return '正在直播'
    case 'expired':
      return session.playback_url ? '可观看回放' : '已结束'
    default:
      return '待开播'
  }
}
```

#### 2.1 API错误处理（HTTP状态码）

**所有API请求必须使用以下错误处理逻辑**：

```typescript
// utils/errorHandler.ts
import type { ApiError } from '@/types/api'

/**
 * 统一错误处理函数
 * @param error API错误对象
 * @param showToast 是否显示Toast提示（默认true）
 */
export const handleApiError = (error: any, showToast = true): void => {
  const code = error?.code || error?.statusCode || 500
  let message = '操作失败，请重试'
  
  // HTTP状态码处理
  switch (code) {
    case 401:
      // Token过期或未认证
      message = '登录已过期，请重新登录'
      if (showToast) {
        uni.showToast({
          title: message,
          icon: 'none',
          duration: 2000,
          success: () => {
            // 跳转登录页（延迟500ms以显示Toast）
            setTimeout(() => {
              uni.navigateTo({ url: '/pages/login/index' })
            }, 500)
          }
        })
      }
      break
      
    case 403:
      // 权限不足
      message = '权限不足，无法访问'
      if (showToast) {
        uni.showToast({ title: message, icon: 'none', duration: 2000 })
      }
      break
      
    case 404:
      // 资源不存在
      message = '请求的资源不存在'
      if (showToast) {
        uni.showToast({ title: message, icon: 'none', duration: 2000 })
      }
      break
      
    case 429:
      // 请求过于频繁
      message = '操作过于频繁，请稍后再试'
      if (showToast) {
        uni.showToast({ title: message, icon: 'none', duration: 2000 })
      }
      break
      
    case 500:
    case 502:
    case 503:
    case 504:
      // 服务器错误
      message = '服务器异常，请稍后重试'
      if (showToast) {
        uni.showToast({
          title: message,
          icon: 'none',
          duration: 2000
        })
      }
      break
      
    case -1:
      // 网络超时或连接失败
      message = '网络连接失败，请检查网络'
      if (showToast) {
        uni.showModal({
          title: '网络异常',
          content: message,
          showCancel: true,
          cancelText: '取消',
          confirmText: '重试',
          success: (res) => {
            if (res.confirm) {
              // 用户点击重试，可以在这里重新发起请求
              // 或者抛出一个特殊的错误码，由调用方处理
            }
          }
        })
      }
      break
      
    default:
      // 业务错误码（2xxx、3xxx、4xxx）
      message = error?.message || '操作失败，请重试'
      if (showToast) {
        uni.showToast({ title: message, icon: 'none', duration: 2000 })
      }
  }
  
  // 开发环境打印错误详情（生产环境不打印）
  if (import.meta.env.DEV) {
    console.error('[API Error]', {
      code,
      message,
      details: error?.details || error,
      timestamp: new Date().toISOString()
    })
  }
}
```

```typescript
const TRUSTED_IMAGE_DOMAINS = [
  'mp.dayilive.com',
  'cdn.dayilive.com',
  'static.dayilive.com'
]

/**
 * 安全的图片URL获取（复用已有的URL校验逻辑）
 * @param url 原始URL
 * @param type 图片类型
 * @returns 安全的图片URL或占位图
 * @example getImageUrl('https://cdn.dayilive.com/cover.jpg', 'cover')
 * @example getImageUrl(null, 'avatar') // 返回默认头像占位图
 */
export function getImageUrl(url: string | undefined | null, type: 'cover' | 'avatar' = 'cover'): string {
  if (!url || typeof url !== 'string' || url.trim() === '') {
    return type === 'cover' ? DEFAULT_COVER_PLACEHOLDER : DEFAULT_AVATAR_PLACEHOLDER
  }
  
  const trimmedUrl = url.trim()
  
  // 特殊处理：禁止SVG data URI（可能包含script标签）
  if (trimmedUrl.toLowerCase().startsWith('data:image/svg')) {
    console.warn('[Security] SVG data URI blocked for XSS protection')
    return type === 'cover' ? DEFAULT_COVER_PLACEHOLDER : DEFAULT_AVATAR_PLACEHOLDER
  }
  
  // 复用已有的URL安全校验（validateUrl函数）
  if (!validateUrl(trimmedUrl)) {
    console.warn('[Security] Blocked unsafe image URL:', trimmedUrl)
    return type === 'cover' ? DEFAULT_COVER_PLACEHOLDER : DEFAULT_AVATAR_PLACEHOLDER
  }
  
  // 生产环境域名白名单校验（图片专用）
  if (import.meta.env.PROD && trimmedUrl.startsWith('http')) {
    try {
      const urlObj = new URL(trimmedUrl)
      const isAllowed = TRUSTED_IMAGE_DOMAINS.some(domain => {
        return urlObj.hostname === domain || urlObj.hostname.endsWith('.' + domain)
      })
      
      if (!isAllowed) {
        console.warn('[Security] Image domain not in whitelist:', urlObj.hostname)
        return type === 'cover' ? DEFAULT_COVER_PLACEHOLDER : DEFAULT_AVATAR_PLACEHOLDER
      }
    } catch {
      return type === 'cover' ? DEFAULT_COVER_PLACEHOLDER : DEFAULT_AVATAR_PLACEHOLDER
    }
  }
  
  // 相对路径转绝对路径
  if (trimmedUrl.startsWith('/')) {
    const baseUrl = import.meta.env.VITE_API_BASE_URL || 'https://mp.dayilive.com'
    return `${baseUrl}${trimmedUrl}`
  }
  
  return trimmedUrl
}
```

```typescript
    'history_id', 'expert_id', 'topic_id'
  ]
  
  // 3. 需要移除查询参数和脱敏路径ID的URL字段
  const urlKeys = ['page_url', 'referer', 'redirect_url', 'current_url']
  
  // 4. 时间戳字段（保留到小时）
  const timestampKeys = [
    'watched_at', 'created_at', 'updated_at', 
    'notified_at', 'subscribed_at'
  ]
  
  // 步骤1：移除敏感字段
  removeKeys.forEach(key => {
    if (key in sanitized) {
      delete sanitized[key]
    }
  })
  
  // 步骤2：UUID脱敏（仅保留前8位）
  uuidKeys.forEach(key => {
    if (key in sanitized && typeof sanitized[key] === 'string') {
      const value = sanitized[key]
      if (value.length >= 8) {
        sanitized[key] = value.substring(0, 8) + '****'
      }
    }
  })
  
  // 步骤3：URL脱敏（移除查询参数 + 路径ID脱敏）
  urlKeys.forEach(key => {
    if (key in sanitized && typeof sanitized[key] === 'string') {
      try {
        const url = new URL(sanitized[key])
        
        // 脱敏路径中的数字ID（如 /users/123456 → /users/***）
        // 脱敏路径中的UUID（如 /sessions/550e8400-... → /sessions/***）
        let pathname = url.pathname
        
        // 替换纯数字ID（如 /123456/ 或 /123456）
        pathname = pathname.replace(/\/\d+(\/|$)/g, '/***$1')
        
        // 替换UUID格式（8-4-4-4-12）
        pathname = pathname.replace(/\/[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}(\/|$)/gi, '/***$1')
        
        // 移除查询参数，仅保留origin + pathname
        sanitized[key] = url.origin + pathname
      } catch {
        // URL解析失败，删除该字段
        delete sanitized[key]
      }
    }
  })
  
  // 步骤4：时间戳脱敏（保留到小时，隐藏分秒）
  timestampKeys.forEach(key => {
    if (key in sanitized && typeof sanitized[key] === 'string') {
      try {
        const date = new Date(sanitized[key])
        // 转换为ISO格式，仅保留到小时（分秒设为00）
        sanitized[key] = date.toISOString().substring(0, 13) + ':00:00Z'
      } catch {
        // 时间戳解析失败，删除该字段
        delete sanitized[key]
      }
    }
  })
  
  return sanitized
}
```

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { getWatchHistory, deleteWatchHistory } from '@/api/watchHistory'
import type { WatchHistory } from '@/types/watchHistory'
import dayjs from 'dayjs'

interface GroupedHistory {
  date: string
  items: WatchHistory[]
}

export const useWatchHistoryStore = defineStore('watchHistory', () => {
  // 状态
  const historyList = ref<WatchHistory[]>([])
  const loading = ref(false)
  const hasMore = ref(true)
  const page = ref(1)
  const size = ref(20)
  
  // 分组后的历史（使用computed缓存）
  const groupedHistory = computed<GroupedHistory[]>(() => {
    return groupHistoryByDate(historyList.value)
  })
  
  // 分组算法
  function groupHistoryByDate(items: WatchHistory[]): GroupedHistory[] {
    const today = dayjs().startOf('day')
    const yesterday = dayjs().subtract(1, 'day').startOf('day')
    
    const groups = {
      today: [] as WatchHistory[],
      yesterday: [] as WatchHistory[],
      earlier: [] as WatchHistory[]
    }
    
    items.forEach(item => {
      const watchedAt = dayjs(item.watched_at)
      if (watchedAt.isSame(today, 'day')) {
        groups.today.push(item)
      } else if (watchedAt.isSame(yesterday, 'day')) {
        groups.yesterday.push(item)
      } else {
        groups.earlier.push(item)
      }
    })
    
    return [
      { date: '今天', items: groups.today },
      { date: '昨天', items: groups.yesterday },
      { date: '更早', items: groups.earlier }
    ].filter(group => group.items.length > 0)
  }
  
  // 加载历史记录
  async function fetchHistory(refresh = false) {
    if (loading.value) return
    
    if (refresh) {
      page.value = 1
      hasMore.value = true
    }
    
    try {
      loading.value = true
      const res = await getWatchHistory({
        page: page.value,
        size: size.value
      })
      
      if (res.code === 200 && res.data) {
        const newItems = res.data.items || []
        
        if (refresh) {
          historyList.value = newItems
        } else {
          historyList.value.push(...newItems)
        }
        
        hasMore.value = newItems.length >= size.value
      }
    } catch (error) {
      console.error('[WatchHistory] Fetch failed:', error)
      throw error
    } finally {
      loading.value = false
    }
  }
  
  // 加载更多
  async function loadMore() {
    if (!hasMore.value || loading.value) return
    
    page.value++
    await fetchHistory(false)
  }
  
  // 删除单条
  async function deleteHistory(historyId: string) {
    try {
      await deleteWatchHistory(historyId)
      historyList.value = historyList.value.filter(item => item.id !== historyId)
    } catch (error) {
      console.error('[WatchHistory] Delete failed:', error)
      throw error
    }
  }
  
  // 清空全部
  async function clearAll() {
    // 调用清空API
    // await clearWatchHistory()
    historyList.value = []
    page.value = 1
    hasMore.value = true
  }
  
  return {
    historyList,
    groupedHistory,
    loading,
    hasMore,
    fetchHistory,
    loadMore,
    deleteHistory,
    clearAll
  }
})
```

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

  const upcomingList = computed(() => {
    return subscriptions.value.filter(sub => {

      return startTime.isAfter(dayjs()) && sub.status === 'scheduled'
    })
  })
  
  // 已通知列表
  const notifiedList = computed(() => {
    return subscriptions.value.filter(sub => {
      return sub.is_notified === true || sub.status === 'live'
    })
  })
  
  // 已过期列表
  const expiredList = computed(() => {
    return subscriptions.value.filter(sub => {
      const endTime = dayjs(sub.end_time)
      return endTime.isBefore(dayjs()) && sub.status === 'ended'
    })
  })
  
  // 未读数量
  const unreadCount = computed(() => ({
    upcoming: upcomingList.value.length,
    notified: notifiedList.value.filter(s => !s.is_read).length,
    expired: 0
  }))
  
  // 加载订阅列表
  async function fetchSubscriptions() {
    try {
      loading.value = true
      const res = await getSubscriptions()
      
      if (res.code === 200 && res.data) {
        subscriptions.value = res.data.items || []
      }
    } catch (error) {
      console.error('[Subscription] Fetch failed:', error)
      throw error
    } finally {
      loading.value = false
    }
  }
  
  // 取消订阅
  async function cancelSubscription(subscriptionId: string) {
    try {
      await removeSubscription(subscriptionId)
      subscriptions.value = subscriptions.value.filter(s => s.id !== subscriptionId)
    } catch (error) {
      console.error('[Subscription] Cancel failed:', error)
      throw error
    }

    // 清除已通知和已过期
    const toRemove = [
      ...notifiedList.value,
      ...expiredList.value
    ]

      await cancelSubscription(sub.id)
    }
  }
  
  return {
    subscriptions,
    upcomingList,
    notifiedList,
    expiredList,
```

### 10.5 轮询管理器 (`src/utils/polling.ts`)

**文件路径**: `src/utils/polling.ts`

**处理方式**: ✅ **新建文件**（项目中不存在）

**平台兼容说明**：
- ✅ **uni-app全平台兼容**（H5/App/小程序）
- ✅ H5平台：使用document和window API（页面可见性检测 + 网络状态检测）
- ✅ App平台：使用uni.onAppShow/onAppHide（应用前后台切换检测）
- ✅ 小程序平台：使用uni.onAppShow/onAppHide（小程序前后台切换检测）

```typescript
/**
 * 轮询管理器（uni-app全平台兼容版）
 * 支持页面可见性检测和网络状态检测
 * 
 * 平台差异处理：
 * - H5: 使用document.visibilityState和navigator.onLine
 * - App/小程序: 使用uni.onAppShow/onAppHide监听前后台切换
 */
export class PollingManager {
  private timer: number | null = null
  private interval: number
  private isActive: boolean = false
  private callback: (() => void) | null = null
  private onAppShowCallback: (() => void) | null = null
  
  constructor(interval: number = 30000) {
    this.interval = interval
  }
  
  start(callback: () => void): void {
    this.callback = callback
    this.isActive = true
    
    // 首次立即执行
    this.poll()
    
    // 监听页面可见性或应用前后台切换
    this.addVisibilityListener()
    
    // 监听网络状态
    this.addNetworkListener()
  }
  
  stop(): void {
    this.isActive = false
    this.callback = null
    
    if (this.timer) {
      clearTimeout(this.timer)
      this.timer = null
    }
    
    this.removeListeners()
  }
  
  private poll(): void {
    if (this.timer) {
      clearTimeout(this.timer)
      this.timer = null
    }
    
    if (!this.isActive || !this.callback) return
    
    // H5平台检查页面可见性
    // #ifdef H5
    if (typeof document !== 'undefined' && document.visibilityState !== 'visible') {
      this.scheduleNextPoll()
      return
    }
    // #endif
    
    // H5平台检查网络状态
    // #ifdef H5
    if (typeof navigator !== 'undefined' && !navigator.onLine) {
      this.scheduleNextPoll()
      return
    }
    // #endif
    
    // 执行回调
    try {
      this.callback()
    } catch (error) {
      console.error('[Polling] Callback error:', error)
    }
    
    // 调度下次轮询
    this.scheduleNextPoll()
  }
  
  private scheduleNextPoll(): void {
    if (!this.isActive) return
    
    this.timer = setTimeout(() => {
      this.poll()
    }, this.interval) as unknown as number
  }
  
  private addVisibilityListener(): void {
    // #ifdef H5
    if (typeof document !== 'undefined') {
      document.addEventListener('visibilitychange', this.handleVisibilityChange)
    }
    // #endif
    
    // #ifdef APP-PLUS || MP-WEIXIN || MP-ALIPAY
    // App和小程序平台：监听应用前后台切换
    this.onAppShowCallback = () => {
      if (this.isActive) {
        this.poll() // 应用回到前台，立即轮询
      }
    }
    uni.onAppShow(this.onAppShowCallback)
    // #endif
  }
  
  private addNetworkListener(): void {
    // #ifdef H5
    if (typeof window !== 'undefined') {
      window.addEventListener('online', this.handleNetworkOnline)
    }
    // #endif
    
    // #ifdef APP-PLUS || MP-WEIXIN || MP-ALIPAY
    // App和小程序平台：使用uni.onNetworkStatusChange
    uni.onNetworkStatusChange((res) => {
      if (res.isConnected && this.isActive) {
        this.poll() // 网络恢复，立即轮询
      }
    })
    // #endif
  }
  
  private removeListeners(): void {
    // #ifdef H5
    if (typeof document !== 'undefined') {
      document.removeEventListener('visibilitychange', this.handleVisibilityChange)
    }
    if (typeof window !== 'undefined') {
      window.removeEventListener('online', this.handleNetworkOnline)
    }
    // #endif
    
    // #ifdef APP-PLUS || MP-WEIXIN || MP-ALIPAY
    // uni-app的onAppShow无法手动移除，会在页面卸载时自动清理
    // 通过isActive标记来控制回调是否执行
    // #endif
  }
  
  private handleVisibilityChange = (): void => {
    // #ifdef H5
    if (typeof document !== 'undefined' && document.visibilityState === 'visible' && this.isActive) {
      this.poll() // 页面重新可见，立即轮询
    }
    // #endif
  }
  
  private handleNetworkOnline = (): void => {
    if (this.isActive) {
      this.poll() // 网络恢复，立即轮询
    }
  }
}
```

  try {

```typescript
    uni.setStorageSync(ENCRYPTED_TOKEN_KEY, encrypted)

  try {
    const encrypted = uni.getStorageSync(ENCRYPTED_TOKEN_KEY)
    if (!encrypted) {
      return uni.getStorageSync('access_token') || null
    }
    
    const decrypted = CryptoJS.AES.decrypt(encrypted, SECRET_KEY)
    return decrypted.toString(CryptoJS.enc.Utf8) || null
  } catch (error) {
    console.error('[Auth] Token decryption failed:', error)
    return uni.getStorageSync('access_token') || null
  }
}
```

#### 3.2 URL协议校验

**所有跳转链接必须校验URL协议，防止javascript:伪协议攻击**：

```typescript
// utils/urlValidator.ts

/**
 * 校验URL是否安全
 * @param url 待校验的URL
 * @returns 是否安全
 */
export const isSafeUrl = (url: string): boolean => {
  if (!url) return false
  
  // 允许的协议白名单
  const allowedProtocols = ['http:', 'https:', 'tel:', 'mailto:']
  
  try {
    const urlObj = new URL(url, window.location.origin)
    return allowedProtocols.includes(urlObj.protocol)
  } catch {
    // URL解析失败，可能是相对路径，允许通过
    return !url.startsWith('javascript:') && !url.startsWith('data:')
  }
}

/**
 * 安全跳转函数
 * @param url 跳转URL
 */
export const safeNavigate = (url: string): void => {
  if (!isSafeUrl(url)) {
    console.warn('[Security] Blocked unsafe URL:', url)
    uni.showToast({ title: '链接不安全，已阻止跳转', icon: 'none' })
    return
  }
  
  // 安全跳转
  if (url.startsWith('http://') || url.startsWith('https://')) {
    // 外部链接
    uni.navigateTo({ url: `/pages/webview/index?url=${encodeURIComponent(url)}` })
  } else {
    // 内部路由
    uni.navigateTo({ url })
  }
}
```
