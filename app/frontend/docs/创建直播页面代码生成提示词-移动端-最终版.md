### 0.1 时间选择器工具函数

为避免代码重复，将时间选择器逻辑提取为公共工具函数：

```typescript
/**
 * 时间选择器工具函数
 * 用于抽屉组件和全屏页面的时间选择功能
 */

export interface DateTimePickerData {
  dateTimeRange: string[][];
  dateTimeValue: number[];
  displayDateTime: string;
}

/**
 * 初始化时间选择器数据（支持选择历史日期和年份）
 * @returns DateTimePickerData
 * 说明：生成5列时间选择器（年、月、日、时、分）
 * 年份范围：过去10年 ~ 未来2年
 */
export function initDateTimePickerData(): DateTimePickerData {
  const years: string[] = [];
  const months: string[] = [];
  const days: string[] = [];
  const hours: string[] = [];
  const minutes: string[] = [];
  
  const now = new Date();
  const currentYear = now.getFullYear();
  const currentMonth = now.getMonth() + 1;
  const currentDay = now.getDate();
  
  // 生成年份：过去10年到未来2年（参考H5端）
  for (let i = currentYear - 10; i <= currentYear + 2; i++) {
    years.push(i + '年');
  }
  
  // 生成月份：1-12月
  for (let i = 1; i <= 12; i++) {
    months.push(i + '月');
  }
  
  // 生成日期：1-31日
  for (let i = 1; i <= 31; i++) {
    days.push(i + '日');
  }
  
  // 生成24小时
  for (let i = 0; i < 24; i++) {
    hours.push(i.toString().padStart(2, '0') + '时');
  }
  
  // 生成分钟（每5分钟一档）
  for (let i = 0; i < 60; i += 5) {
    minutes.push(i.toString().padStart(2, '0') + '分');
  }
  
  const dateTimeRange = [years, months, days, hours, minutes];
  
  // 设置默认时间为当前时间+30分钟
  now.setMinutes(now.getMinutes() + 30);
  const roundedMinutes = Math.ceil(now.getMinutes() / 5) * 5;
  now.setMinutes(roundedMinutes);
  
  const defaultYearIdx = 10; // 当前年份索引（过去10年 + 当前年）
  const defaultMonthIdx = currentMonth - 1;
  const defaultDayIdx = currentDay - 1;
  const defaultHour = now.getHours();
  const defaultMinute = Math.floor(now.getMinutes() / 5);
  
  const dateTimeValue = [defaultYearIdx, defaultMonthIdx, defaultDayIdx, defaultHour, defaultMinute];
  const displayDateTime = getDisplayDateTime(dateTimeRange, dateTimeValue);
  
  return {
    dateTimeRange,
    dateTimeValue,
    displayDateTime
  };
}

/**
 * 获取显示的时间文本
 * @param dateTimeRange 时间范围数组
 * @param dateTimeValue 选中的索引数组
 * @returns 格式化的时间字符串
 */
export function getDisplayDateTime(
  dateTimeRange: string[][],
  dateTimeValue: number[]
): string {
  const [yearIdx, monthIdx, dayIdx, hourIdx, minuteIdx] = dateTimeValue;
  const yearStr = dateTimeRange[0][yearIdx] || '';
  const monthStr = dateTimeRange[1][monthIdx] || '';
  const dayStr = dateTimeRange[2][dayIdx] || '';
  const hourStr = dateTimeRange[3][hourIdx] || '';
  const minuteStr = dateTimeRange[4][minuteIdx] || '';
  return `${yearStr}${monthStr}${dayStr} ${hourStr}${minuteStr}`;
}

/**
 * 将选择器值转换为ISO时间字符串
 * @param dateTimeValue 选中的索引数组
 * @returns ISO格式的时间字符串
 */
export function dateTimeValueToISO(dateTimeValue: number[]): string {
  const [yearIdx, monthIdx, dayIdx, hourIdx, minuteIdx] = dateTimeValue;
  
  const now = new Date();
  const currentYear = now.getFullYear();
  
  // 计算实际年份（过去10年到未来2年）
  const year = currentYear - 10 + yearIdx;
  const month = monthIdx; // 月份0-11
  const day = dayIdx + 1; // 日期1-31
  
  const targetDate = new Date(year, month, day, hourIdx, minuteIdx * 5, 0, 0);
  
  // ... 其余时区处理代码
  return targetDate.toISOString();
}
```

**脚本部分**：

```vue
<script setup lang="ts">
import { ref, computed, watch, onMounted } from 'vue';
import { useRoomStore } from '@/store/room';
import { useSessionStore } from '@/store/session';
import { useAuthStore } from '@/store/auth';
import { 
  initDateTimePickerData, 
  getDisplayDateTime, 
  dateTimeValueToISO 
} from '@/utils/datetime';

const props = defineProps<{
  visible: boolean;
}>();

const emit = defineEmits<{
  'update:visible': [value: boolean];
  'success': [roomId: string];
}>();

const roomStore = useRoomStore();
const sessionStore = useSessionStore();
const authStore = useAuthStore();

// 表单数据
const formData = ref({
  title: '',
  description: '',
  start_time: '',
  cover_url: ''
});

// 错误信息
const errors = ref({
  title: '',
  start_time: ''
});

// 提交状态
const isSubmitting = ref(false);

// 动画控制
const showMask = ref(false);
const showContent = ref(false);
const scrollTop = ref(0);

// 抽屉高度（适配手机屏幕）
const drawerHeight = computed(() => {
  return '70vh'; // 使用70vh确保在手机上有足够空间
});

// 时间选择器数据（5列：年、月、日、时、分）
const dateTimeRange = ref<string[][]>([[], [], [], [], []]);
const dateTimeValue = ref<number[]>([0, 0, 0, 0, 0]);
const displayDateTime = ref('请选择开始时间');

/**
 * 初始化时间选择器（使用公共工具函数）
 */
const initDateTimeRange = () => {
  const pickerData = initDateTimePickerData();
  dateTimeRange.value = pickerData.dateTimeRange;
  dateTimeValue.value = pickerData.dateTimeValue;
  displayDateTime.value = pickerData.displayDateTime;
  formData.value.start_time = dateTimeValueToISO(pickerData.dateTimeValue);
};

/**
 * 时间选择器change事件
 */
const handleDateTimeChange = (e: any) => {
  dateTimeValue.value = e.detail.value;
  displayDateTime.value = getDisplayDateTime(dateTimeRange.value, dateTimeValue.value);
  formData.value.start_time = dateTimeValueToISO(dateTimeValue.value);
  errors.value.start_time = '';
};

/**
 * 表单验证
 */
const validateForm = (): boolean => {
  let isValid = true;
  errors.value.title = '';
  errors.value.start_time = '';
  
  if (!formData.value.title.trim()) {
    errors.value.title = '请输入直播标题';
    isValid = false;
  }
  
  if (!formData.value.start_time) {
    errors.value.start_time = '请选择开始时间';
    isValid = false;
  }
  
  // 验证时间是否在未来
  if (formData.value.start_time) {
    const startTime = new Date(formData.value.start_time);
    const now = new Date();
    if (startTime <= now) {
      errors.value.start_time = '开始时间必须在当前时间之后';
      isValid = false;
    }
  }
  
  return isValid;
};

/**
 * 是否可以提交
 */
const canSubmit = computed(() => {
  return formData.value.title.trim() && formData.value.start_time && !isSubmitting.value;
});

/**
 * 创建直播
 */
const handleCreate = async () => {
  if (!validateForm()) {
    return;
  }
  
  // 检查登录状态
  if (!authStore.isAuthenticated) {
    uni.showToast({
      title: '请先登录',
      icon: 'none'
    });
    return;
  }
  
  isSubmitting.value = true;
  
  try {
    // 第一步：创建Room
    console.log('[CreateLiveDrawer] 开始创建直播间...');
    const roomResult = await roomStore.addNewRoom({
      title: formData.value.title.trim(),
      description: formData.value.description.trim() || '',
      cover_url: formData.value.cover_url || '/logo.png',
      is_private: false
    });
    
    if (!roomResult.success) {
      throw new Error(roomResult.message || '创建直播间失败');
    }
    
    console.log('[CreateLiveDrawer] 直播间创建成功，room_id:', roomResult.room_id);
    
    // 第二步：创建Session
    console.log('[CreateLiveDrawer] 开始设置直播时间...');
    const sessionResult = await sessionStore.createSession(roomResult.room_id!, {
      start_time: new Date(formData.value.start_time)
    });
    
    if (!sessionResult.success) {
      console.warn('[CreateLiveDrawer] 设置直播时间失败，但直播间已创建');
    }
    
    // 成功提示
    uni.showToast({
      title: '创建成功',
      icon: 'success'
    });
    
    // 重置表单
    resetForm();
    
    // 关闭抽屉
    handleClose();
    
    // 触发成功回调
    emit('success', roomResult.room_id!);
    
    // 跳转到直播管理页面
    setTimeout(() => {
      uni.navigateTo({
        url: '/pages/app/live-manage/list'
      });
    }, 500);
    
  } catch (error: any) {
    console.error('[CreateLiveDrawer] 创建失败:', error);
    uni.showToast({
      title: error.message || '创建失败，请重试',
      icon: 'none',
      duration: 2000
    });
  } finally {
    isSubmitting.value = false;
  }
};

/**
 * 重置表单
 */
const resetForm = () => {
  formData.value = {
    title: '',
    description: '',
    start_time: '',
    cover_url: ''
  };
  errors.value = {
    title: '',
    start_time: ''
  };
  initDateTimeRange();
};

/**
 * 关闭抽屉
 */
const handleClose = () => {
  showMask.value = false;
  showContent.value = false;
  
  setTimeout(() => {
    emit('update:visible', false);
  }, 300);
};

/**
 * 监听visible变化，控制动画
 */
watch(() => props.visible, (newVal) => {
  if (newVal) {
    setTimeout(() => {
      showMask.value = true;
      showContent.value = true;
    }, 50);
  } else {
    showMask.value = false;
    showContent.value = false;
  }
});

/**
 * 组件挂载时初始化
 */
onMounted(() => {
  initDateTimeRange();
});
</script>
```

const handleSearchBrands = async () => {

    if (response.data && response.data.items) {

 * 表单验证

const validateForm = (): boolean => {
  let isValid = true;
  errors.value.title = '';
  errors.value.description = '';
  errors.value.start_time = '';
  
  if (!formData.value.title.trim()) {
    errors.value.title = '请输入直播标题';
    isValid = false;
  }
  
  if (!formData.value.description.trim()) {
    errors.value.description = '请输入直播简介';
    isValid = false;
  }
  
  if (!formData.value.start_time) {
    errors.value.start_time = '请选择开始时间';
    isValid = false;
  }
  
  // 验证时间是否在未来
  if (formData.value.start_time) {
    const startTime = new Date(formData.value.start_time);
    const now = new Date();
    if (startTime <= now) {
      errors.value.start_time = '开始时间必须在当前时间之后';
      isValid = false;
    }
  }
  
  return isValid;
};

 * 是否可以提交

         formData.value.description.trim() && 
         formData.value.start_time && 
         !isSubmitting.value;

      description: formData.value.description.trim(),
      cover_url: formData.value.cover_url || '/logo.png',
      is_private: formData.value.is_private
    });
    
    if (!roomResult.success) {
      throw new Error(roomResult.message || '创建直播间失败');
    }
    
    // 创建场次
    const sessionResult = await sessionStore.createSession(roomResult.room_id!, {
      start_time: new Date(formData.value.start_time)
    });
    
    if (!sessionResult.success) {
      console.warn('设置直播时间失败，但直播间已创建');
    }
    
    uni.showToast({
      title: '创建成功',
      icon: 'success'
    });
    
    setTimeout(() => {
      uni.navigateBack();
    }, 500);
    
  } catch (error: any) {
    console.error('创建失败:', error);
    uni.showToast({
      title: error.message || '创建失败，请重试',
      icon: 'none',
      duration: 2000
    });
  } finally {
    isSubmitting.value = false;
  }
};

/**
 * 返回

};

      showCancel: false,
      success: () => {
        uni.navigateBack();
      }
    });

  }
