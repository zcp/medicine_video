### 1.1 问题分析

**原设计方案**：

**存在的问题**：
- ❌ ActionSheet无法展示专家头像、详细信息
- ❌ 无法搜索和筛选
- ❌ 无法查看专家详情
- ❌ 用户体验差

### 1.2 优化方案：复用现有页面

**方案A：创建专门的选择页面（推荐）** ⭐

**文件路径**：
- `src/pages/app/expert/select.vue` - 专家选择页面
- `src/pages/app/brand/select.vue` - 品牌选择页面

**实现方式**：
1. 复用现有的列表页面逻辑和组件
2. 通过 `mode="select"` 参数区分选择模式和浏览模式
3. 点击卡片后通过事件通信返回选中数据

**页面跳转**：
```typescript
// 跳转到专家选择页面
uni.navigateTo({
  url: '/pages/app/expert/select',
  events: {
    // 监听选择事件
    selectExpert: (data: ExpertData) => {
      selectedExpert.value = data;
      formData.value.featured_expert_id = data.id;
    }
  }
});
```

**选择页面返回数据**：
```typescript
// 在选择页面中
const handleSelectExpert = (expert: Expert) => {
  // 通过事件通道返回数据
  const eventChannel = uni.getOpenerEventChannel();
  eventChannel.emit('selectExpert', {
    id: expert.id,
    name: expert.name,
    title: expert.title,
    department: expert.department,

  // 返回上一页

```

**方案B：直接跳转到现有列表页面**

**实现方式**：
1. 在现有的列表页面添加选择模式支持
2. 通过URL参数 `?mode=select` 控制
3. 选择模式下点击卡片触发选择而非跳转详情

**优缺点对比**：

| 方案 | 优点 | 缺点 |
|------|------|------|
| 方案A | 逻辑清晰、不影响现有页面 | 需要创建新页面 |
| 方案B | 代码复用度高、无需新页面 | 现有页面逻辑复杂化 |

**推荐**：采用方案A，创建专门的选择页面。

## 二、安全增强方案

### 2.1 图片上传安全增强

#### 2.1.1 文件头验证（Magic Number）

```typescript
/**
 * 🔒 验证图片文件头（Magic Number）
 * 防止伪造文件扩展名
 */
const validateImageMagicNumber = async (filePath: string): Promise<boolean> => {
  try {
    const fs = uni.getFileSystemManager();
    
    // 读取文件前8个字节
    const buffer = await new Promise<ArrayBuffer>((resolve, reject) => {
      fs.readFile({
        filePath,
        position: 0,
        length: 8,
        success: (res) => resolve(res.data),
        fail: reject
      });
    });
    
    const bytes = new Uint8Array(buffer);
    
    // 检查文件头
    // JPEG: FF D8 FF
    if (bytes[0] === 0xFF && bytes[1] === 0xD8 && bytes[2] === 0xFF) {
      return true;
    }
    
    // PNG: 89 50 4E 47 0D 0A 1A 0A
    if (bytes[0] === 0x89 && bytes[1] === 0x50 && bytes[2] === 0x4E && bytes[3] === 0x47) {
      return true;
    }
    
    // WebP: 52 49 46 46 ... 57 45 42 50
    if (bytes[0] === 0x52 && bytes[1] === 0x49 && bytes[2] === 0x46 && bytes[3] === 0x46) {
      if (bytes.length >= 12 && bytes[8] === 0x57 && bytes[9] === 0x45 && bytes[10] === 0x42 && bytes[11] === 0x50) {
        return true;
      }
    }
    
    return false;
  } catch (error) {
    console.error('验证文件头失败:', error);
    return false;
  }
};

/**
 * 🔒 增强的图片上传函数
 */
const handleUploadCover = async () => {
  try {
    const res = await uni.chooseImage({
      count: 1,
      sizeType: ['compressed'],
      sourceType: ['album', 'camera']
    });
    
    if (res.tempFilePaths && res.tempFilePaths.length > 0) {
      const filePath = res.tempFilePaths[0];
      
      // 🔒 安全验证1：文件大小
      const fileInfo = await uni.getFileInfo({ filePath });
      const maxSize = 10 * 1024 * 1024; // 10MB
      
      if (fileInfo.size > maxSize) {
        uni.showToast({
          title: '图片大小不能超过10MB',
          icon: 'none'
        });
        return;
      }
      
      // 🔒 安全验证2：文件扩展名
      const allowedExtensions = ['.jpg', '.jpeg', '.png', '.webp'];
      const fileExtension = filePath.substring(filePath.lastIndexOf('.')).toLowerCase();
      
      if (!allowedExtensions.includes(fileExtension)) {
        uni.showToast({
          title: '仅支持JPG、PNG、WEBP格式',
          icon: 'none'
        });
        return;
      }
      
      // 🔒 安全验证3：文件头验证（新增）
      const isValidImage = await validateImageMagicNumber(filePath);
      if (!isValidImage) {
        uni.showToast({
          title: '图片格式不正确，请重新选择',
          icon: 'none'
        });
        return;
      }
      
      // 验证通过
      formData.value.cover_url = filePath;
    }
  } catch (error) {
    console.error('选择图片失败:', error);
    uni.showToast({
      title: '选择图片失败',
      icon: 'none'
    });
  }
};
```

```typescript
/**
 * 🔒 验证图片尺寸比例
 * 建议16:9，允许一定误差
 */
const validateImageAspectRatio = async (filePath: string): Promise<boolean> => {
  return new Promise((resolve) => {
    uni.getImageInfo({
      src: filePath,
      success: (res) => {
        const ratio = res.width / res.height;
        const targetRatio = 16 / 9;
        const tolerance = 0.1; // 10%误差
        
        const isValid = Math.abs(ratio - targetRatio) / targetRatio <= tolerance;
        
        if (!isValid) {
          uni.showToast({
            title: '建议使用16:9比例的图片',
            icon: 'none'
          });
        }
        
        resolve(isValid);
      },
      fail: () => {
        resolve(false);
      }
    });
  });
};
```

### 2.3 草稿数据加密

```typescript
/**
 * 🔒 加密草稿数据
 * 使用简单的Base64编码（可升级为AES加密）
 */
const saveDraft = () => {
  try {
    const draftData = {
      ...formData.value,
      timestamp: Date.now()
    };
    
    // 简单加密（Base64）
    const encrypted = btoa(JSON.stringify(draftData));
    
    uni.setStorageSync('live_draft_encrypted', encrypted);
    
    uni.showToast({
      title: '草稿已保存',
      icon: 'success'
    });
  } catch (error) {
    console.error('保存草稿失败:', error);
  }
};

/**
 * 🔒 解密草稿数据
 */
const loadDraft = () => {
  try {
    const encrypted = uni.getStorageSync('live_draft_encrypted');
    if (encrypted) {
      const decrypted = JSON.parse(atob(encrypted));
      
      // 检查草稿是否过期（7天）
      const maxAge = 7 * 24 * 60 * 60 * 1000;
      if (Date.now() - decrypted.timestamp > maxAge) {
        uni.removeStorageSync('live_draft_encrypted');
        return null;
      }
      
      return decrypted;
    }
  } catch (error) {
    console.error('加载草稿失败:', error);
  }
  return null;
};
```

```vue
  if (hours > 0) return `${hours}小时前`;
  if (minutes > 0) return `${minutes}分钟前`;
  return '刚刚';
```

### 3.2 网络错误重试机制

```typescript
/**
 * 🔄 网络请求重试机制
 */
const retryRequest = async <T>(
  requestFn: () => Promise<T>,
  maxRetries: number = 3,
  delay: number = 1000
): Promise<T> => {
  let lastError: any;
  
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await requestFn();
    } catch (error: any) {
      lastError = error;
      
      // 如果是网络错误，等待后重试
      if (error.message?.includes('网络') || error.statusCode === 0) {
        if (i < maxRetries - 1) {
          await new Promise(resolve => setTimeout(resolve, delay * (i + 1)));
          continue;
        }
      }
      
      // 其他错误直接抛出
      throw error;
    }
  }
  
  throw lastError;
};

// 使用示例
const handleCreate = async () => {
  try {
    // 创建房间（带重试）
    const roomRes = await retryRequest(() => createRoom(roomPayload), 3, 1000);
    
    // 上传封面（带重试）
    if (formData.value.cover_url) {
      await retryRequest(
        () => uploadCoverToServer(roomId, formData.value.cover_url),
        2,
        500
      );
    }
    
    // 创建场次（带重试）
    await retryRequest(() => createSession(roomId, sessionPayload), 3, 1000);
    
  } catch (error) {
    // 错误处理
  }
};
```

```typescript
/**
 * 🔄 创建失败回滚策略

const handleCreate = async () => {
  let roomId: string | null = null;

  try {
    // 1. 创建房间
    const roomRes = await createRoom(roomPayload);

            } catch (e) {

      // 房间创建失败
```
