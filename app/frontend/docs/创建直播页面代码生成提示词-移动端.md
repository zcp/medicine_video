```typescript
// 时间选择器数据
const dateTimeRange = ref<string[][]>([[], [], []]);
const dateTimeValue = ref<number[]>([0, 0, 0]);
const displayDateTime = ref('请选择开始时间');

/**
 * 初始化时间选择器数据
 */
const initDateTimeRange = () => {
  const dates: string[] = [];
  const hours: string[] = [];
  const minutes: string[] = [];
  
  // 生成未来30天的日期
  for (let i = 0; i < 30; i++) {
    const date = new Date();
    date.setDate(date.getDate() + i);
    const month = date.getMonth() + 1;
    const day = date.getDate();
    const weekDay = ['周日', '周一', '周二', '周三', '周四', '周五', '周六'][date.getDay()];
    dates.push(`${month}月${day}日 ${weekDay}`);
  }
  
  // 生成24小时
  for (let i = 0; i < 24; i++) {
    hours.push(i.toString().padStart(2, '0') + '时');
  }
  
  // 生成分钟（每5分钟一档）
  for (let i = 0; i < 60; i += 5) {
    minutes.push(i.toString().padStart(2, '0') + '分');
  }
  
  dateTimeRange.value = [dates, hours, minutes];
  
  // 设置默认时间为当前时间+30分钟，并向上取整到最近的5分钟
  const now = new Date();
  now.setMinutes(now.getMinutes() + 30);
  const roundedMinutes = Math.ceil(now.getMinutes() / 5) * 5;
  now.setMinutes(roundedMinutes);
  
  const defaultHour = now.getHours();
  const defaultMinute = Math.floor(now.getMinutes() / 5);
  
  dateTimeValue.value = [0, defaultHour, defaultMinute];
  updateDisplayDateTime();
  updateFormDateTime();
};

/**
 * 更新显示的时间文本
 */
const updateDisplayDateTime = () => {
  const [dateIdx, hourIdx, minuteIdx] = dateTimeValue.value;
  const dateStr = dateTimeRange.value[0][dateIdx] || '';
  const hourStr = dateTimeRange.value[1][hourIdx] || '';
  const minuteStr = dateTimeRange.value[2][minuteIdx] || '';
  displayDateTime.value = `${dateStr} ${hourStr}${minuteStr}`;
};

/**
 * 更新表单中的ISO时间字符串
 */
const updateFormDateTime = () => {
  const [dateIdx, hourIdx, minuteIdx] = dateTimeValue.value;
  
  const targetDate = new Date();
  targetDate.setDate(targetDate.getDate() + dateIdx);
  targetDate.setHours(hourIdx);
  targetDate.setMinutes(minuteIdx * 5);
  targetDate.setSeconds(0);
  targetDate.setMilliseconds(0);
  
  formData.value.start_time = targetDate.toISOString();
};

/**
 * 时间选择器change事件
 */
const handleDateTimeChange = (e: any) => {
  dateTimeValue.value = e.detail.value;

  updateFormDateTime();

  // 可以在这里添加列联动逻辑，例如日期变化时调整小时范围

/**
 * 上传封面图片
 * 🔒 安全验证：文件大小、文件类型
 */
const handleUploadCover = async () => {
  try {
    const res = await uni.chooseImage({
      count: 1,
      sizeType: ['compressed'],
      sourceType: ['album', 'camera']
    });
    
    if (res.tempFilePaths && res.tempFilePaths.length > 0) {
      const filePath = res.tempFilePaths[0];
      
      // 🔒 安全验证1：检查文件大小（最大10MB）
      try {
        const fileInfo = await uni.getFileInfo({ filePath });
        const maxSize = 10 * 1024 * 1024; // 10MB
        
        if (fileInfo.size > maxSize) {
          uni.showToast({
            title: '图片大小不能超过10MB',
            icon: 'none',
            duration: 2000
          });
          return;
        }
      } catch (error) {
        console.error('获取文件信息失败:', error);
      }
      
      // 🔒 安全验证2：检查文件扩展名
      const allowedExtensions = ['.jpg', '.jpeg', '.png', '.webp'];
      const fileExtension = filePath.substring(filePath.lastIndexOf('.')).toLowerCase();
      
      if (!allowedExtensions.includes(fileExtension)) {
        uni.showToast({
          title: '仅支持JPG、PNG、WEBP格式',
          icon: 'none',
          duration: 2000
        });
        return;
      }
      
      // 验证通过，设置封面
      formData.value.cover_url = filePath;
    }
  } catch (error) {
    console.error('选择图片失败:', error);
    uni.showToast({
      title: '选择图片失败',
      icon: 'none'
    });
  }

/**
 * 🔒 验证图片文件头（Magic Number）
 */
const validateImageMagicNumber = async (filePath: string): Promise<boolean> => {
  try {
    const fs = uni.getFileSystemManager();
    
    // 读取文件前8个字节
    const buffer: ArrayBuffer = await new Promise((resolve, reject) => {
      fs.readFile({
        filePath,
        position: 0,
        length: 8,
        success: (res: any) => resolve(res.data),
        fail: reject
      });
    });
    
    const bytes = new Uint8Array(buffer);
    
    // JPEG: FF D8 FF
    if (bytes[0] === 0xFF && bytes[1] === 0xD8 && bytes[2] === 0xFF) {
      return true;
    }
    
    // PNG: 89 50 4E 47 0D 0A 1A 0A
    if (bytes[0] === 0x89 && bytes[1] === 0x50 && bytes[2] === 0x4E && bytes[3] === 0x47) {
      return true;
    }
    
    // WebP: 52 49 46 46 ... 57 45 42 50
    if (bytes[0] === 0x52 && bytes[1] === 0x49 && bytes[2] === 0x46 && bytes[3] === 0x46) {
      if (bytes.length >= 12 && bytes[8] === 0x57 && bytes[9] === 0x45 && bytes[10] === 0x42 && bytes[11] === 0x50) {
        return true;
      }
    }
    
    return false;
  } catch (error) {
    console.error('验证文件头失败:', error);
    return false;
  }
};

 * 🔄 网络请求重试机制

const retryRequest = async <T>(
  requestFn: () => Promise<T>,
  maxRetries: number = 3,
  delay: number = 1000
): Promise<T> => {
  let lastError: any;
  
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await requestFn();
    } catch (error: any) {
      lastError = error;
      
      // 如果是网络错误，等待后重试
      if (error.message?.includes('网络') || error.statusCode === 0) {
        if (i < maxRetries - 1) {
          await new Promise(resolve => setTimeout(resolve, delay * (i + 1)));
          continue;
        }
      }
      
      // 其他错误直接抛出
      throw error;
    }
  }
  
  throw lastError;
};

import { getToken } from '@/utils/auth';

/**
 * 上传直播间封面
 * @param roomId 房间ID
 * @param filePath 本地文件路径
 * @returns Promise<any>
 */
export const uploadCover = (roomId: string, filePath: string): Promise<any> => {
  const token = getToken();
  if (!token) {
    return Promise.reject(new Error('未登录，无法上传封面'));
  }
  
  const BASE_URL = import.meta.env.VITE_API_BASE_URL || '/api/v1';
  
  return new Promise((resolve, reject) => {
    uni.uploadFile({
      url: `${BASE_URL}/rooms/${roomId}/cover`,
      filePath: filePath,
      name: 'file',
      header: {
        'Authorization': `Bearer ${token}`
      },
      success: (res) => {
        if (res.statusCode === 200) {
          try {
            const data = JSON.parse(res.data);
            resolve(data);
          } catch (error) {
            reject(new Error('解析响应失败'));
          }
        } else {
          reject(new Error(`上传失败: ${res.statusCode}`));
        }
      },
      fail: (error) => {
        reject(error);
      }
    });
  });
};
```

#### 1.3 权限规则

**根据后端API文档**（`直播核心功能设计文档_v6_深度融合最终版.md` 第880行）：

```markdown
认证要求: 所有以下API接口都需要JWT Token认证
```

**权限要求**：
- ✅ **允许创建直播**：所有已登录且状态为`NORMAL`的用户
- ❌ **不允许创建直播**：未登录用户、被封禁用户（`BANNED`）、已删除用户（`DELETED`）、待审核用户（`PENDING_REVIEW`）

**实现逻辑**（`src/utils/auth.ts:126-143`）：
```typescript
export const canCreateLive = (): boolean => {
  const authStore = useAuthStore();
  
  // 1. 检查是否已登录
  if (!authStore.isAuthenticated || !authStore.user) {
    return false;
  }
  
  // 2. 检查用户状态（被封禁、删除、待审核的用户不能创建直播）
  if (authStore.user.status !== 'NORMAL') {
    return false;
  }
  
  // 3. 所有已登录且状态正常的用户都可以创建直播
  return true;
};
```

**说明**：
- 后端API文档**没有要求**特定角色权限（如ADMIN、MODERATOR）
- 后端API文档**没有要求**主播认证或专家认证
- 只要求用户已登录（有JWT Token）且状态正常即可
