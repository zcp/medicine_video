```python
    async def set_live_room_categories(
        self, db: AsyncSession, room_id: UUID, body: LiveRoomCategoriesSetRequest,
        current_user_id: UUID, role: str,
    ) -> LiveRoomCategoriesSetResponse:
        """批量设置某直播间的分类（仅 ADMIN 或 SUPERADMIN——分类为全局受控词汇）。"""
        # 权限检查：仅 ADMIN 或 SUPERADMIN

        check_admin_permission(role)
        # 校验 category_ids
        requested = set(body.category_ids)
        if requested:
            result = await db.execute(select(Category).where(Category.id.in_(body.category_ids)))
            existing = result.scalars().all()
            active_ids = {c.id for c in existing if c.is_active}
            if requested - active_ids:
                raise InvalidParameterException("部分分类ID不存在或未启用")
        lrcs = await crud_content_management.set_live_room_categories(
            db, room_id, body.category_ids, body.mode, body.primary_category_id
        )
        categories_payload = [
            RoomCategoryItem(**CategoryItem.model_validate(lrc.category).model_dump(), is_primary=lrc.is_primary)
            for lrc in lrcs
        ]
        await db.commit()
        return LiveRoomCategoriesSetResponse(
            code=200, message="success",
            data={"room_id": str(room_id), "mode": body.mode, "categories": categories_payload},
            timestamp=datetime.now(),
        )
```

```python
@admin_rooms_router.delete("/rooms/batch")
async def batch_delete_rooms(
    body: BatchDeleteRequest,
    current_user: dict = Depends(get_current_user),
):
    """批量删除直播间（独立事务 + 直播中保护）"""
    user_id = uuid.UUID(current_user["user_id"])
    role = current_user.get("role")
    user_id_log = str(user_id)[:8]

    try:
        check_admin_permission(role)
    except PermissionDeniedException:
        logger.warning(f"权限不足: user_id={user_id_log}, role={role}")
        return JSONResponse(status_code=403, content=error_response(code=3003, message="权限不足"))

    results = []
    deleted_count = 0
    skipped_count = 0

    for room_id in body.room_ids:
        try:
            # 每个房间使用独立的 DB session
            async with AsyncSessionLocal() as room_db:
                room_service = RoomService(db=room_db)
                await room_service.delete_room(
                    room_id=room_id, user_id=user_id, role=role,
                    force=body.force  # ← 传入 force 参数
                )
                results.append(BatchDeleteResultItem(room_id=room_id, status="deleted"))
                deleted_count += 1
        except Exception as e:
            reason = str(e)
            logger.warning(f"批量删除失败: room_id={room_id}, user_id={user_id_log}, error={reason}")
            results.append(BatchDeleteResultItem(room_id=room_id, status="skipped", reason=reason))
            skipped_count += 1

    return success_response(data=BatchDeleteResponse(
        total_requested=len(body.room_ids),
        deleted=deleted_count,
        skipped=skipped_count,
        results=results,
    ).model_dump())
```

> **注意**：`delete_room()` 需要新增 `force` 参数（见 §4.4）。`AsyncSessionLocal` 需从 `app.database` 导入（确认该文件中已有此导出）。

```python
            db=self.db, page=page, size=size, role=role,
            q=q, sort=sort, status=status,
            created_after=created_after, created_before=created_before,
        )

        if not rooms_data:
            return {"total": 0, "page": page, "size": size, "items": []}

        room_ids = [r["room_id"] for r in rooms_data]

        # Step 2: 批量聚合查询（3 次）
        category_map = await batch_get_category_names(self.db, room_ids)
        expert_map = await batch_get_expert_names(self.db, room_ids)
        tag_map = await batch_get_tag_names(self.db, room_ids)
        session_count_map = await batch_get_session_counts(self.db, room_ids)

        # Step 3: 组装
        items = []
        for r in rooms_data:
            rid = r["room_id"]
            items.append(RoomSummaryItem(
                room_id=rid,
                title=r["title"],
                cover_url=r.get("cover_url"),
                is_private=r.get("is_private", False),
                user_id=r["user_id"],
                created_at=r["created_at"],
                updated_at=r["updated_at"],
                live_status=r.get("live_status"),
                current_session_id=r.get("current_session_id"),
                latest_session_start_time=r.get("session_start_time"),
                category_names=category_map.get(rid, []),
                expert_names=expert_map.get(rid, []),
                tag_names=tag_map.get(rid, []),
                session_count=session_count_map.get(rid, 0),
            ))

        return {"total": total, "page": page, "size": size, "items": [i.model_dump(mode="json") for i in items]}
```

#### 4.5.1 `get_rooms_with_latest_session()` — 主查询

```python
async def get_rooms_with_latest_session(
    db: AsyncSession,
    page: int = 1, size: int = 20,
    role: Optional[str] = None,
    q: Optional[str] = None,
    sort: Optional[str] = None,
    status: Optional[List[str]] = None,
    created_after: Optional[str] = None,
    created_before: Optional[str] = None,
) -> Tuple[List[dict], int]:
    """
    管理摘要主查询：房间 + 最新 Session 状态。
    返回 (rooms_data_list, total)
    每个 dict 含：room_id, title, cover_url, is_private, user_id, created_at, updated_at,
                  live_status, current_session_id, session_start_time
    """
    # 窗口函数子查询：每个房间取最新 session（按 start_time DESC）
    latest_subq = (
        select(
            LiveSession.room_id,
            LiveSession.status.label("live_status"),
            LiveSession.id.label("session_id"),
            LiveSession.start_time.label("session_start_time"),
            func.row_number()
                .over(partition_by=LiveSession.room_id, order_by=LiveSession.start_time.desc())
                .label("rn"),
        )
        .subquery()
    )

    query = (
        select(
            LiveRoom.id.label("room_id"),
            LiveRoom.title,
            LiveRoom.cover_url,
            LiveRoom.is_private,
            LiveRoom.user_id,
            LiveRoom.created_at,
            LiveRoom.updated_at,
            latest_subq.c.live_status,
            latest_subq.c.session_id.label("current_session_id"),
            latest_subq.c.session_start_time,
        )
        .outerjoin(
            latest_subq,
            and_(latest_subq.c.room_id == LiveRoom.id, latest_subq.c.rn == 1),
        )
    )

    # 权限过滤（Admin 看全部）
    if role not in ("ADMIN", "SUPERADMIN"):
        raise PermissionDeniedException("需要管理员权限")

    # 搜索过滤
    if q:
        uuid_pattern = re.compile(
            r'^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$',
            re.IGNORECASE,
        )
        if uuid_pattern.match(q):
            query = query.where(LiveRoom.id == q)
        else:
            query = query.where(LiveRoom.title.ilike(f"%{q}%"))

    # status 筛选
    if status:
        valid_statuses = {e.value for e in LiveSessionStatus}
        invalid = [s for s in status if s.lower() not in valid_statuses]
        if invalid:
            raise InvalidParameterException(f"无效的状态值: {invalid}", code=4001)
        status_enum_values = [LiveSessionStatus[s.upper()] for s in status]
        query = query.where(
            exists().where(
                and_(
                    LiveSession.room_id == LiveRoom.id,
                    LiveSession.status.in_(status_enum_values),
                )
            )
        )

    # 时间范围筛选
    if created_after:
        dt_after = datetime.fromisoformat(created_after) if isinstance(created_after, str) else created_after
        query = query.where(LiveRoom.created_at >= dt_after)
    if created_before:
        from datetime import datetime
        dt_before = datetime.fromisoformat(created_before) if isinstance(created_before, str) else created_before
        query = query.where(LiveRoom.created_at <= dt_before)

    # 计数
    count_query = select(func.count()).select_from(query.subquery())
    count_result = await db.execute(count_query)
    total = count_result.scalar()

    # 排序
    if sort:
        sort_parts = sort.split(":")
        if len(sort_parts) == 2:
            field_name, direction = sort_parts[0], sort_parts[1].upper()
            if field_name == "start_time" and direction in ("ASC", "DESC"):
                col = latest_subq.c.session_start_time
                query = query.order_by(col.desc().nulls_last() if direction == "DESC" else col.asc().nulls_last())
            elif hasattr(LiveRoom, field_name) and direction in ("ASC", "DESC"):
                col = getattr(LiveRoom, field_name)
                query = query.order_by(col.desc() if direction == "DESC" else col.asc())
            else:
                query = query.order_by(LiveRoom.created_at.desc())
        else:
            query = query.order_by(LiveRoom.created_at.desc())
    else:
        query = query.order_by(LiveRoom.created_at.desc())

    # 分页
    query = query.offset((page - 1) * size).limit(size)

    result = await db.execute(query)
    rows = result.all()

    rooms_data = []
    for row in rows:
        rooms_data.append({
            "room_id": row.room_id,
            "title": row.title,
            "cover_url": row.cover_url,
            "is_private": row.is_private,
            "user_id": row.user_id,
            "created_at": row.created_at,
            "updated_at": row.updated_at,
            "live_status": row.live_status,
            "current_session_id": row.current_session_id,
            "session_start_time": row.session_start_time,
        })

    return rooms_data, total
```

```python
async def batch_get_category_names(db: AsyncSession, room_ids: List[uuid.UUID]) -> dict:
    """返回 {room_id: [category_name, ...]}"""
    if not room_ids:
        return {}
    result = await db.execute(
        select(LiveRoomCategory.room_id, Category.name)
        .join(Category, LiveRoomCategory.category_id == Category.id)
        .where(LiveRoomCategory.room_id.in_(room_ids), Category.is_active == True)
        .order_by(LiveRoomCategory.room_id)
    )
    mapping = {}
    for row in result.all():
        mapping.setdefault(row.room_id, []).append(row.name)
    return mapping


async def batch_get_expert_names(db: AsyncSession, room_ids: List[uuid.UUID]) -> dict:
    """返回 {room_id: [expert_name, ...]}，取全部 session 的专家去重"""
    if not room_ids:
        return {}
    result = await db.execute(
        select(LiveSession.room_id, Expert.name)
        .distinct()
        .join(LiveSessionExpert, LiveSession.id == LiveSessionExpert.session_id)
        .join(Expert, LiveSessionExpert.expert_id == Expert.id)
        .where(LiveSession.room_id.in_(room_ids))
        .order_by(LiveSession.room_id)
    )
    mapping = {}
    for row in result.all():
        mapping.setdefault(row.room_id, []).append(row.name)
    return mapping


async def batch_get_tag_names(db: AsyncSession, room_ids: List[uuid.UUID]) -> dict:
    """返回 {room_id: [tag_name, ...]}，取全部 session 的标签去重"""
    if not room_ids:
        return {}
    result = await db.execute(
        select(LiveSession.room_id, Tag.name)
        .distinct()
        .join(SessionTag, LiveSession.id == SessionTag.session_id)
        .join(Tag, SessionTag.tag_id == Tag.id)
        .where(LiveSession.room_id.in_(room_ids))
        .order_by(LiveSession.room_id)
    )
    mapping = {}
    for row in result.all():
        mapping.setdefault(row.room_id, []).append(row.name)
    return mapping


async def batch_get_session_counts(db: AsyncSession, room_ids: List[uuid.UUID]) -> dict:
    """返回 {room_id: count}"""
    if not room_ids:
        return {}
    result = await db.execute(
        select(LiveSession.room_id, func.count(LiveSession.id))
        .where(LiveSession.room_id.in_(room_ids))
        .group_by(LiveSession.room_id)
    )
    return {row.room_id: row[1] for row in result.all()}
```

```python
    async def batch_set_live_room_categories(
        self,
        db: AsyncSession,
        room_ids: List[UUID],
        category_ids: List[UUID],
        mode: str,
        current_user_id: UUID,
        role: str,
    ) -> dict:
        """批量设置直播间分类（独立事务，部分成功语义）"""
        # 权限检查：仅 ADMIN/SUPERADMIN
        check_admin_permission(role)

        # 一次性校验所有 category_ids
        if category_ids:
            result = await db.execute(
                select(Category).where(Category.id.in_(category_ids), Category.is_active == True)
            )
            existing_ids = {c.id for c in result.scalars().all()}
            if set(category_ids) - existing_ids:
                raise InvalidParameterException("部分分类ID不存在或未启用")

        results = []
        success_count = 0
        failed_count = 0

        for room_id in room_ids:
            try:
                # 复用单房间分类设置逻辑
                body = LiveRoomCategoriesSetRequest(
                    category_ids=category_ids,
                    primary_category_id=category_ids[0] if category_ids else None,
                    mode=mode,
                )
                await self.set_live_room_categories(
                    db=db, room_id=room_id, body=body,
                    current_user_id=current_user_id, role=role,
                )
                results.append({"room_id": str(room_id), "status": "success"})
                success_count += 1
            except Exception as e:
                await db.rollback()  # 清理失败房间的脏事务，确保下一个房间不受影响
                results.append({
                    "room_id": str(room_id),
                    "status": "failed",
                    "reason": str(e),
                })
                failed_count += 1

        return {
            "total_requested": len(room_ids),
            "success": success_count,
            "failed": failed_count,
            "results": results,
        }
```

> **注意**：`set_live_room_categories` 内部会 `await db.commit()`。批量循环中每次 commit 后同一个 session 仍然可用（PostgreSQL + asyncpg 中 commit 不会关闭 session）。异常分支已包含 `await db.rollback()` 以清理失败房间的脏事务。
