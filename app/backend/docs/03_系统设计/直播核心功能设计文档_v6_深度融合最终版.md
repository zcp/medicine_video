```sql
-- LiveCore Service - 完整数据库 Schema (V3.2)


CREATE TYPE live_session_status AS ENUM (
    'scheduled',    -- [新增] 已计划，待开播
    'live', 'finished', 'processing', 'ready', 'error'
);

-- 表 1: live_rooms (持久化容器)
CREATE TABLE live_rooms (
 -- 核心标识 (在应用层通过 uuid.uuid4() 生成)
    id UUID PRIMARY KEY,
    
    -- 用户ID，通过JWT Token应用层验证保证引用完整性
    -- 由于采用微服务架构，user_id无法跨数据库外键关联，通过JWT Token验证用户存在性
    user_id UUID NOT NULL,
   
    -- 【新增】自引用外键，用于实现“主会场-分会场”结构
    parent_room_id UUID NULL REFERENCES live_rooms(id) ON DELETE SET NULL,

    title VARCHAR(100) NOT NULL,
    description TEXT,
    cover_url VARCHAR(255),
    stream_key VARCHAR(255) NOT NULL UNIQUE,
    is_private BOOLEAN DEFAULT false,
    record_by_default BOOLEAN DEFAULT true,

    -- 分类信息
    category_id UUID NULL, -- [待办] 初期允许为空。未来在分类模块实现后，应添加 FOREIGN KEY REFERENCES categories(id)。
 
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP

);
CREATE INDEX idx_live_rooms_user_id ON live_rooms(user_id);
COMMENT ON TABLE live_rooms IS '直播房间表，存储可复用的持久化信息和推流密钥。';

-- 表 2: live_sessions (瞬时事件)
CREATE TABLE live_sessions (

    -- 核心标识 (在应用层通过 uuid.uuid4() 生成)
    id UUID PRIMARY KEY,

    room_id UUID NOT NULL REFERENCES live_rooms(id) ON DELETE CASCADE,

    -- 【核心修改】status 不再有默认值，由应用层在创建时明确指定
    status live_session_status NOT NULL,

    start_time TIMESTAMPTZ NOT NULL,
    end_time TIMESTAMPTZ,

-- 【为未来扩展预留的字段】
    video_id UUID NULL UNIQUE, -- [待办] 用于关联到未来的 VOD 媒资表 (videos)。当前阶段可不使用，在回放功能开发时启用。
    -- 【新增】回放地址，创建时默认为 NULL，可由系统自动生成，也可人工修改
    playback_url VARCHAR(1024) NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

COMMENT ON TABLE live_sessions IS '直播场次表，记录每一次具体的直播事件及其生命周期（含可选的回放地址）。';

-- 表 3: session_statistics (场次统计)
CREATE TABLE session_statistics (
   -- 核心标识 (在应用层通过 uuid.uuid4() 生成)
    id UUID PRIMARY KEY,

    session_id UUID NOT NULL UNIQUE REFERENCES live_sessions(id) ON DELETE CASCADE,
    peak_viewer_count INT DEFAULT 0,
    total_viewer_count BIGINT DEFAULT 0,
    total_like_count BIGINT DEFAULT 0,
    total_share_count BIGINT DEFAULT 0,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE session_statistics IS '单场直播的统计数据表，与 live_sessions 表一一对应。';

-- 索引
CREATE INDEX idx_live_rooms_on_user_id ON live_rooms(user_id);
CREATE INDEX idx_live_sessions_on_room_id ON live_sessions(room_id);
CREATE INDEX idx_live_sessions_on_status ON live_sessions(status);
CREATE INDEX idx_live_sessions_on_video_id ON live_sessions(video_id);


-- ====== 增量：Tab & 留言 ======

CREATE TYPE live_room_message_user_role AS ENUM (
    'REGULAR',

    'SUPERADMIN'
);

-- 迁移：为 live_room_tabs.content_type 创建 ENUM 类型
CREATE TYPE live_room_tab_content_type AS ENUM (
    'text',
    'image',
    'mixed'
);

-- 留言表
CREATE TABLE live_room_messages (
    id UUID PRIMARY KEY,
    room_id UUID NOT NULL
        REFERENCES live_rooms(id)
        ON DELETE CASCADE,
    session_id UUID NULL
        REFERENCES live_sessions(id)
        ON DELETE SET NULL,

    -- 存储 users.public_id (来自 JWT sub)
    user_id UUID NOT NULL,

    -- [学院派迁移] 字段类型从 VARCHAR(32) 迁移到 ENUM
    user_role live_room_message_user_role NOT NULL,

    content TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    is_deleted BOOLEAN NOT NULL DEFAULT FALSE,

-- 索引
CREATE INDEX idx_live_room_messages_room_created
```

## **开发规范**

### **1. 编码与项目规范**

* **常量 (Constant)**: 使用全大写下划线命名法 (UPPER\_SNAKE\_CASE)，例如 `MAX_CONNECTIONS`。

### **2. 日志与异常处理规范**

#### **2.1 日志规范**

##### **2.1.1 日志级别**
* `ERROR`: 关键系统错误、导致业务失败的异常。必须立即关注。
* `WARNING`: 潜在的问题或警告信息，不影响当前流程但需关注。
* `INFO`: 记录重要的业务操作节点，如用户登录、创建直播间等。
* `DEBUG`: 用于开发和调试阶段，记录详细的程序运行信息。

##### **2.1.2 日志格式**
每一条日志记录都应包含以下标准字段：
* 时间戳 (ISO 8601 格式)
* 日志级别 (如: INFO)
* 模块名 (如: `routers.rooms`)
* 函数名
* 行号
* 消息内容
* 异常堆栈 (仅在记录异常时包含)

##### **2.1.3 日志内容**
应记录但不限于以下关键信息：
* 系统启动与关闭事件。
* 用户认证操作（登录/登出），需注意脱敏。
* 核心业务操作的入口和结果（如创建/更新/删除房间）。
* 所有捕获到的异常信息。
* 关键性能监控数据（如 API 耗时）。

##### **2.1.4 日志管理与存储**
* **集中管理**: 使用 ELK Stack (Elasticsearch, Logstash, Kibana) 进行日志的统一收集、存储和查询。
* **存储策略**:
    * 日志文件按日期进行分割和归档。
    * 对用户密码、密钥等所有敏感信息必须进行脱敏处理。

##### **2.1.5 日志脱敏规范（安全要求）**
* **脱敏原则**: 所有日志输出必须经过脱敏处理，防止敏感信息泄露。
* **脱敏范围**:
    * **数据库连接URL**: 移除连接字符串中的密码部分，仅保留服务器、端口、数据库名、用户名等非敏感信息。
    * **JWT Token**: 移除完整的Token字符串，避免在日志中暴露认证凭证。
    * **环境变量**: 移除包含 `PASSWORD`、`SECRET`、`KEY`、`TOKEN` 等关键词的环境变量值。
    * **Authorization头**: 移除HTTP请求头中的完整认证信息。
* **实现方式**:
    * 使用统一的日志脱敏工具函数 `sanitize_log_message()` 对所有日志消息进行处理。
    * 使用 `SanitizedFormatter` 类作为日志格式化器，自动对所有日志输出进行脱敏。
    * 在系统启动时调用 `setup_sanitized_logging()` 配置脱敏日志系统。
* **示例**:
    ```python
    # 脱敏前
    logger.info(f"尝试连接数据库: postgresql://user:password123@localhost:5432/dbname")
    
    # 脱敏后
    logger.info(f"尝试连接数据库: postgresql://user:***@localhost:5432/dbname")
    ```

#### **2.2 异常处理规范**

##### **2.2.1 异常分类**
* **系统异常**: 系统级错误（如数据库连接失败、中间件故障）。
* **业务异常**: 不符合业务规则的正常操作（如余额不足、库存不够）。
* **参数异常**: 用户输入参数不符合格式或校验规则。
* **权限异常**: 用户无权访问特定资源或执行特定操作。

##### **2.2.2 异常处理原则**
* **统一处理**: 实现统一的异常处理中间件 (Exception Handling Middleware) 来捕获所有未处理的异常，避免程序崩溃。
* **明确类型**: 使用自定义的、继承自 `Exception` 的异常类来区分不同的异常情况。
* **详细日志**: 捕获到任何异常时，都必须记录详细的错误日志，包含完整的异常堆栈。
* **格式统一**: 返回给客户端的错误响应必须遵循 `2.1. 通用响应结构` 的格式：    
比如：
{
  "code": 2005,
  "message": "无法删除：存在关联数据，请先解除品牌/专家等关联或联系管理员",
  "data": {
    "resource_id": "room_uuid_123"
  },
  "timestamp": "2025-07-08T14:40:00Z"
}


* **避免吞没**: 严禁捕获异常后不做任何处理（`except: pass`）。
* **优雅降级**: 在可能的情况下，对系统异常进行优雅降级处理，保证核心功能的可用性。

##### **2.2.3 异常处理流程**
1.  在业务代码中**捕获**可预见的异常。
2.  将原始异常**记录**到日志系统。
3.  将原始异常**转换**为对应的自定义业务异常类型。
4.  由统一的异常处理中间件捕获所有异常，并**返回**统一格式的错误响应。
5.  在必要时（如文件句柄、数据库连接），使用 `finally` 块**清理**资源。

### **3. 配置与安全规范**

#### **3.1 配置管理原则**

##### **3.1.1 环境变量管理**
* **统一配置源**: 所有配置项必须通过环境变量管理，禁止在代码中硬编码敏感信息。
* **配置模板**: 提供 `.env.example` 文件作为配置模板，包含所有必需的环境变量及说明。
* **环境隔离**: 
    * 本地开发环境使用 `.env` 文件（不提交到版本控制）。
    * Docker部署环境使用 `.env.docker` 或 `.env.production` 文件（通过 `env_file` 注入）。
    * 测试环境使用 `.env.example` 文件。
* **配置验证**: 
    * 生产环境启动时必须验证所有必需配置项（数据库密码、JWT密钥等）。
    * 开发环境启动时对缺失配置发出友好警告，但不阻止启动。

##### **3.1.2 敏感信息保护**
* **禁止硬编码**: 
    * ❌ 禁止在代码中硬编码数据库密码、JWT密钥、API密钥等敏感信息。
    * ❌ 禁止在代码中设置不安全的默认值（如 `"CHANGE_ME"`、`"my-key"`）。
* **环境变量读取**: 
    * ✅ 所有敏感配置必须从环境变量读取，默认值为空字符串 `""`。
    * ✅ 生产环境必须设置所有敏感环境变量，否则启动失败。
* **配置集中管理**: 
    * ✅ 所有配置统一在 `app/core/config.py` 中管理，避免重复定义。
    * ✅ 数据库配置、JWT配置、CORS配置等统一使用 `Settings` 类。

##### **3.1.3 CORS配置外部化**
* **配置方式**: CORS允许的源列表通过 `CORS_ORIGINS` 环境变量配置，支持逗号分隔的多个域名。
* **格式示例**:
    * 开发环境: `CORS_ORIGINS=http://localhost:5175,http://localhost:3000`
    * 生产环境: `CORS_ORIGINS=https://yourdomain.com,https://www.yourdomain.com`
* **安全要求**: 
    * ⚠️ 生产环境禁止使用通配符 `*`，必须指定具体的允许域名。
    * ⚠️ 开发环境可以使用 `*`，但建议明确指定本地开发域名。

#### **3.2 安全配置要求**

##### **3.2.1 数据库安全**
* **密码要求**: 
    * 生产环境必须设置强密码（至少16字符），禁止使用默认密码。
    * 开发环境允许使用默认密码，但会发出警告提示。
* **连接安全**: 
    * 数据库连接URL不得在日志中完整打印，仅打印服务器、端口、数据库名、用户名等非敏感信息。
    * 使用统一的配置类管理数据库连接，避免在多个位置重复定义。

##### **3.2.2 JWT认证安全**
* **密钥要求**: 
    * JWT密钥长度必须至少32个字符（推荐64字符以上）。
    * 生产环境必须设置强密钥，禁止使用默认密钥 `"my-key"`。
    * 密钥生成方式: `openssl rand -hex 32` 或 `python -c "import secrets; print(secrets.token_hex(32))"`
* **密钥管理**: 
    * JWT密钥必须从环境变量 `JWT_SECRET_KEY` 读取。
    * 密钥必须与用户服务保持一致，确保Token验证通过。
* **算法选择**: 使用 `HS256` 算法进行Token签名。

##### **3.2.3 配置验证机制**
* **生产环境验证**: 
    * 启动时强制验证所有必需配置项。
    * 验证失败时抛出异常，阻止应用启动。
    * 验证项包括：数据库密码、JWT密钥长度、CORS配置、DEBUG模式等。
* **开发环境验证**: 
    * 启动时对缺失或不安全的配置发出警告。
    * 提供友好的提示信息，引导开发者正确配置。
    * 不阻止应用启动，但记录警告日志。

     * **日志脱敏**: 所有日志输出自动脱敏，防止敏感信息泄露。
     * **健康检查**: 提供健康检查端点，支持K8s/Docker健康探针。
  * **PostgreSQL 数据库**: 系统的持久化存储中心。
     * **连接安全**: 数据库连接信息通过环境变量配置，不在日志中暴露密码。

## 7\. 数据库设计

### 7.1. 设计原则

  * **双表模型**: `live_rooms` (配置) 与 `live_sessions` (事件) 分离。
  * **UUID 主键**: 所有表主键使用 `UUID`，保证全局唯一性。

### 7.3. V6 增量字段与索引：external_room_id 与 playback_url_hash

> 在不修改既有字段语义的前提下，V6 通过 **DDL 增量** 的方式，为批量导入幂等 / 去重能力引入两个新字段与对应索引。旧数据在这两个字段为 NULL 时不会触发任何新约束。

#### 7.3.1 live_rooms.external_room_id

- 作用：存储外部系统的直播间 ID（例如第三方平台导出 CSV 中的「直播间ID」列），用于在批量导入时跨文件、跨批次复用同一直播间。
- 约束：同一 user_id 下 external_room_id 必须唯一；允许为 NULL（不参与唯一约束）。

```sql
ALTER TABLE live_rooms
  ADD COLUMN IF NOT EXISTS external_room_id VARCHAR(64) NULL;

CREATE UNIQUE INDEX IF NOT EXISTS uq_live_rooms_user_external_room
  ON live_rooms (user_id, external_room_id)
  WHERE external_room_id IS NOT NULL;
```

#### 7.3.2 live_sessions.playback_url_hash

- 作用：对 `playback_url` 做规范化（至少 `strip` 空白）后计算得到的哈希值（推荐使用 SHA-256 十六进制字符串），用于在 Import / Batch Import 场景下按「room + playback_url」维度实现幂等。
- 约束：同一 room_id 下 playback_url_hash 必须唯一；允许为 NULL（不参与唯一约束）。
- 维护：当 PATCH session 清空 `playback_url`（`null` 或空字符串）时，须同步将 `playback_url_hash` 置为 `None`。

```sql
ALTER TABLE live_sessions
  ADD COLUMN IF NOT EXISTS playback_url_hash VARCHAR(128) NULL;

CREATE UNIQUE INDEX IF NOT EXISTS uq_live_sessions_room_playback_hash
  ON live_sessions (room_id, playback_url_hash)
  WHERE playback_url_hash IS NOT NULL;
```

> 数据库迁移时，只需执行上述 ALTER TABLE / CREATE INDEX 语句。普通房间 / 场次创建接口如果不为这两个字段赋值（即保持 NULL），将不会受到唯一约束的影响。

  * **功能描述**: 删除一个直播房间。Service 在删除前先调用 CRUD 的 `delete_room_related(room_id)` 按依赖顺序删除所有关联数据（场次专家、品牌房间、专题房间、留言、Tab、收藏、场次等），再调用 `remove(room)` 删除房间并提交。推流与直播间管理分离，允许在 live 状态下删除。
  * **成功响应** (`200 OK`):

    ```json
      },

    }
    ```

    ```json

    }
    ```
  * **实现流程描述**:
    1.  定义 FastAPI 路由函数，接收 `room_id: UUID` 和数据库会话 `db`。

    4.  **应用层级联删除**: 调用 `crud_room.delete_room_related(db, room_id)` 按依赖顺序删除关联数据（场次专家、品牌房间、专题房间、留言、Tab、收藏、场次等），再调用 `crud_room.remove(db, room)` 删除房间并提交。

    6.  构建并返回操作成功的简单响应。

  * **说明**: 推流与直播间管理分离，不再因「正在直播」禁止删除。详细删除顺序参见《删除直播间-应用层级联删除-实现说明.md》。

### **6.4 内部 API 规范 (SRS 回调)**

本章节详细定义由媒体服务器（SRS）回调的内部 API 接口，这些接口是连接媒体层和业务逻辑层的桥梁，负责驱动直播生命周期的自动管理。

#### **1. `POST /internal/srs/on_publish` (推流发布与会话激活)**

* **职责**:
    验证推流凭证 (`stream_key`) 的有效性，并激活或创建一个直播会话 (`LiveSession`)，以正式记录一场直播的开始。

* **核心逻辑**:
    1.  SRS 在接收到客户端推流时，携带 `stream_key` 等信息回调此接口。
    2.  后端服务根据 `stream_key` 验证其所属的直播间 (`LiveRoom`) 是否存在。若无效，则返回 `403` 拒绝推流。
    3.  验证通过后，系统会检查该直播间是否存在一个“计划中”(`scheduled`) 的会话。
        * **场景A (计划内直播)**: 如果存在，系统会将该会话的状态更新为 `live`，并将其 `start_time` 更新为当前的服务器时间，以精确记录实际开播时刻。
        * **场景B (即兴直播)**: 如果不存在，系统会立即创建一个新的会话，状态直接设为 `live`，`start_time` 设为当前服务器时间。
    4.  处理成功后，向 SRS 返回 `200 OK`，允许推流继续。



#### **2. `POST /internal/srs/on_unpublish` (推流结束与后台任务派发)**

* **职责**:
    接收推流结束事件，将直播会话标记为已完成，并**触发**后续的、耗时的媒体处理任务。

* **核心逻辑与实现方案**:

    1.  **即时响应 (通用逻辑)**:
        * SRS 在检测到推流断开时回调此接口。
        * 后端服务立即查找当前 `live` 状态的直播会话，将其状态更新为 `finished`，并记录 `end_time`。

    2.  **后台任务派发 (核心)**:
        * 在完成状态更新后，接口会立即通过 `.delay()` 调用一个预定义的 Celery 任务 (`post_stream_processing_task`)，并将 `session_id` 等必要信息作为参数传递。
        * 任务消息将被发送至 **Redis** 消息队列，API 接口的调用会**立即返回**，从而保证对 SRS 的快速响应。

    3.  **任务执行架构：原生异步 (Task Execution Architecture: Native Async)**:
        * **核心技术**: 本项目采用 Celery 结合 **Redis** 作为消息中间件，并配置 **`gevent`** 作为其执行池（Execution Pool），以实现原生的高性能异步任务处理。
        * **工作原理**: Celery Worker 本身是异步的，它利用 `gevent` 的协程（Coroutines）能力，可以在单个进程内高并发地处理大量 I/O 密集型任务。当一个任务执行到 `await`（如等待数据库响应）时，Worker 不会阻塞，而是会立刻切换去执行另一个就绪的任务，从而极大地提升了系统的吞吐能力。
        * **任务定义**: 所有的 Celery 任务函数（如 `post_stream_processing_task`）都将被定义为**异步函数 (`async def`)**。这使得在任务内部可以直接、自然地使用 `await` 关键字来调用项目中的其他异步函数（例如异步的 CRUD 操作）。

    4.  **最终响应**:
        * 接口在任务被成功派发到 Redis 队列后，会立即向 SRS 返回 `200 OK`。

#### **3. `post_stream_processing_task` 任务核心逻辑 (由 Celery 执行)**

* **职责**:
    本节定义由 Celery 的原生异步 Worker 执行的后台任务 (`post_stream_processing_task`) 的核心业务逻辑。

* **核心逻辑**:
    1.  **进入 `processing` 状态**: 任务开始执行时，应**立即**将数据库中对应 `session_id` 的记录状态从 `finished` **更新**为 `processing`。
    2.  **执行核心任务**: 执行实际的耗时操作，如视频转码、分析、生成封面等。
    3.  **进入最终状态 (`ready` 或 `error`)**:
        * **场景A (处理成功)**: 如果所有任务顺利完成，在最后一步将该会话的状态从 `processing` **更新**为 `ready`。
           * 如果 `live_sessions.playback_url` 当前为 `NULL`，则根据 7.2 的默认规则生成回放地址，并写入该字段；
           * 如果 `playback_url` 已经存在（例如人工提前配置了第三方回放地址），任务不得覆盖该值，只更新 `status`。

        * **场景B (处理失败)**: 如果在处理过程中发生无法恢复的错误，应捕获异常，并将该会话的状态从 `processing` **更新**为 `error`。

### 7.2. 回放URL生成规范 (更新版)

所有视频回放地址 (`playback_url`) 将由后端**写入 `live_sessions.playback_url` 字段**，并遵循以下规则：

#### 1）URL 基本格式

`<Base_URL>` + `<Storage_Path>`

- `<Base_URL>`：系统的媒体资源访问基地址，应配置为环境变量（例如 `https://your-cdn.com` 或 `https://your-domain.com`），指向 Nginx 或 CDN。
- `<Storage_Path>`：指向经过转码处理后、适合回放的 MP4 文件。推荐格式（默认生成算法）为：

`/media/video_{video_id}/mp4/video_{video_id}_transcoded_{quality}_{timestamp}.mp4`

> 注意：以上是系统推荐的**默认生成规则**，真实字符串会被写入 `live_sessions.playback_url` 并持久化保存。

#### 2）字段语义与生成时机

- `live_sessions.playback_url` 是一个 **可空字段**：
  - 创建场次时，**必须为 NULL**；
  - 直播结束并完成转码后，由后台任务（例如 `post_stream_processing_task`）自动写入默认回放地址；
  - 允许在后续通过管理接口手动修改该字段（例如接入第三方点播系统时替换为外部回放链接）。

#### 3）自动生成的条件（推荐）

后台任务在满足以下条件时，才自动生成并写入回放地址：

1. `status` 字段为 `'ready'`（转码完成且可正常播放）；  
2. `video_id` 字段不为 NULL；  
3. 当前 `playback_url` 为空（避免覆盖人工设置的地址）。
在其他任何状态下（如 `live`, `processing`, `error`, `finished`），`playback_url` 字段将返回 `null`，前端客户端应据此判断视频是否可回放。

满足上述条件时，系统将：

1. 根据 `video_id` 按“默认生成规则”构建完整的 `playback_url` 字符串；
2. 将其写入 `live_sessions.playback_url`；
3. 同步更新 `updated_at`。

#### 4）API 返回约定

- `GET /api/v1/sessions/{session_id}` 中的 `data.playback_url` 字段，**直接取自 `live_sessions.playback_url`**；
- 在典型流程下，该字段由系统自动生成；  
- 若 `playback_url` 仍为 `NULL`，则对外返回 `null`（例如：还在直播、转码中、或尚未配置回放地址）。

## 9\. 身份认证与授权体系

### 9.1 认证架构概述

LiveCore Service采用JWT Token认证机制，与用户服务集成实现统一的身份验证：

- **认证方式**: JWT Token (JSON Web Token)
- **认证流程**: 用户通过用户服务进行SSO登录，获得JWT Token后访问LiveCore Service
- **权限控制**: 1）所有登录用户都可以访问所有直播房间和场次，支持基于 `public_id` 的可选过滤
               2）登录用户只能修改删除直接创建的直播间和直播session（通过比对 `room.user_id == public_id`）
- **Token验证**: 使用与用户服务相同的JWT密钥进行Token验证
- **安全机制**: 支持Token过期检查、签名验证、用户权限验证
- **应用层验证**: 通过JWT Token验证用户存在性，从Token中提取 `public_id`，保证引用完整性
- **数据一致性**: JWT Token有效即证明用户存在，Token中的 `public_id` 可直接用于数据库操作，无需额外验证

### 9.2 认证授权架构

```

│             │    │  (SSO + JWT)    │    │     SSO      │

       │                     │

        ┌────────────┼────────────┐
        │            │            │

**认证组件说明：**
1. **JWT验证模块** (`app/core/auth.py`): 负责Token验证和用户信息提取
2. **认证中间件** (`app/core/deps.py`): 提供认证依赖注入

5. **架构分层**: Service层负责权限验证，CRUD层专注于数据访问，避免重复验证
```

### 9.3 认证授权规范

#### 9.3.1 认证要求
**所有API接口都需要JWT Token认证：**
- 请求头必须包含：`Authorization: Bearer <JWT_TOKEN>`
- Token由用户服务颁发，包含用户身份信息
- 未认证的请求返回401状态码

#### 9.3.2 权限控制
**用户数据隔离原则：**
- 所有登录用户都可以查看所有直播房间和场次
- 支持基于 `public_id` 的可选查询过滤
- 用户只能修改/删除自己的房间和场次（通过比对 `room.user_id == public_id`）
- 所有涉及用户数据的操作都必须验证用户权限
- 通过JWT Token中的 `public_id` 进行权限验证（`public_id = UUID(current_user.get("public_id"))`）
- 留言展示名 `user_display_name` 仅作为展示用途，从已认证的 JWT 中组合 `nickname`/`username`/`email` 得出，并作为快照写入留言记录；不改变任何权限判断逻辑，也不作为用户存在性的唯一依据。

### 9.4 用户认证流程

#### 9.4.1 JWT Token验证流程
```python
def verify_user_authentication(token: str) -> Dict:
    """
    验证用户JWT Token
    :param token: JWT Token字符串
    :return: 用户信息字典
    """
    try:
        # 1. 验证Token签名和格式
        payload = jwt.decode(
            token, 
            JWT_SECRET_KEY, 
            algorithms=[JWT_ALGORITHM]
        )
        
        # 2. 检查Token过期时间
        if payload.get('exp') and time.time() > payload['exp']:
            raise HTTPException(status_code=401, detail="认证凭证已过期")
        
        # 3. 提取用户信息
        user_info = {
            "public_id": payload.get("public_id"),
            "username": payload.get("username"),
            "role": payload.get("role", "user"),
            "permissions": payload.get("permissions", [])
        }
        
        # 4. 验证必要字段
        if not user_info["public_id"]:
            raise HTTPException(status_code=401, detail="Token中缺少用户公开标识(public_id)")
        
        return user_info
        
    except jwt.InvalidTokenError:
        raise HTTPException(status_code=401, detail="认证凭证无效")
    except Exception as e:
        raise HTTPException(status_code=401, detail=f"认证验证失败: {str(e)}")
```

#### 9.4.2 用户权限验证流程
```python
def verify_user_permission(public_id: str, resource_id: str, resource_type: str = "room") -> bool:
    """
    验证用户对资源的访问权限
    :param public_id: 用户公开标识（从JWT提取的public_id）
    :param resource_id: 资源ID
    :param resource_type: 资源类型（room/session）
    :return: 是否有权限访问
    """
    try:
        if resource_type == "room":
            # 验证直播房间权限
            room = get_live_room(resource_id)
            return room and room.user_id == public_id
        elif resource_type == "session":
            # 验证直播场次权限（通过room_id关联）
            session = get_live_session(resource_id)
            if session and session.room_id:
                room = get_live_room(session.room_id)
                return room and room.user_id == public_id
            return False
        else:
            return False
    except Exception:
        return False
```

### 9.5 JWT认证安全

### 9.7 权限验证架构设计

#### 9.7.1 分层权限验证原则

**架构设计原则：**
- **Service层职责**: 负责业务逻辑和权限验证，确保用户只能访问和操作自己的资源
- **CRUD层职责**: 专注于数据访问操作，不包含权限验证逻辑
- **避免重复验证**: 权限验证只在Service层进行，避免在CRUD层重复验证

**权限验证流程：**
1. **API层**: 验证JWT Token，提取用户公开标识（`public_id = UUID(current_user.get("public_id"))`）
2. **Service层**: 进行权限验证，确保用户对资源的访问权限（使用 `public_id` 进行比对）
3. **CRUD层**: 执行纯粹的数据访问操作，不涉及权限验证

**实现示例：**
```python
# Service层权限验证示例
async def update_room_info(self, room_id: uuid.UUID, room_update: LiveRoomUpdate, public_id: uuid.UUID) -> LiveRoom:
    # 1. 验证房间是否存在且属于当前用户
    room = await crud_room.get(db=self.db, room_id=room_id)
    if not room:
        raise RoomNotFoundException()
    
    # 2. 权限验证：确保房间属于当前用户
    if room.user_id != public_id:
        raise PermissionDeniedException("无权修改该房间")
    
    # 3. 推流与直播间管理分离，允许在 live 状态下修改，不再进行 is_live 检查
    
    # 4. 调用CRUD层执行更新（不包含权限验证）
    updated_room = await crud_room.update(db=self.db, db_obj=room, obj_in=room_update)
    return updated_room
```

#### 9.7.2 user_id引用完整性保证

**设计原理：**
- JWT Token由用户服务颁发，包含真实用户的 `public_id`
- Token有效即证明用户存在，无需跨服务验证
- 通过Token提取的 `public_id`（`UUID(current_user.get("public_id"))`）直接用于数据库操作

**实现流程：**
1. 用户登录 → 用户服务验证 → 颁发JWT Token（包含 `public_id`）
2. LiveCore Service验证Token → 提取 `public_id = UUID(current_user.get("public_id"))`
3. 使用 `public_id` 创建/查询数据 → 保证引用完整性

**优势：**
- 无需跨服务查询验证用户存在性
- 性能优异，减少网络调用
- 符合微服务架构最佳实践

### **附录 A: 使用 FFmpeg 进行推流测试**

本附录旨在为开发者和测试人员提供使用 FFmpeg 工具向本系统进行 RTMP 推流的详细指南和常见示例。

#### **1. 推流地址格式**

所有推流请求都应遵循以下 RTMP URL 格式。推流地址由多个部分组成，其中最关键的是从后端 API 获取的 `stream_key`。

**标准格式**:
`rtmp://<srs_ip>:<port>/<app>/<stream_key>`

  * **`<srs_ip>`**: 您的 SRS 媒体服务器的公网 IP 地址或域名。
  * **`<port>`**: SRS 监听的 RTMP 端口，默认为 `1935`。
  * **`<app>`**: 您在 SRS 中配置的应用名称，通常为 `live`。
  * **`<stream_key>`**: **推流密钥**。这是推流的唯一凭证，必须通过调用后端的 `POST /api/v1/rooms` 接口创建直播间来获取。每个直播间拥有唯一的 `stream_key`。

#### **2. 核心推流命令**

FFmpeg 是一个功能强大的命令行工具。以下是一个基础的推流命令模板：

```bash
ffmpeg [输入参数] -i [输入源] [编码参数] -f flv [推流地址]
```
  * **`[输入参数]`**: 控制输入源的参数，例如 `-re` 表示以原生帧率读取文件，模拟真实直播。
  * **`-i [输入源]`**: 指定您的输入内容，可以是一个本地视频文件、一个摄像头设备、或桌面画面。
  * **`[编码参数]`**: 设置视频和音频的编码方式。例如，使用 `-c:v copy -c:a copy` 表示直接复制原始编码，不进行转码，可以极大地降低 CPU 消耗。
  * **`-f flv`**: 强制指定输出格式为 FLV，这是 RTMP 协议的标准容器格式。
  * **`[推流地址]`**: 上一步中定义的标准格式 RTMP URL。

#### **3. 常见推流场景示例**

##### **场景 A: 推送本地视频文件 (最常用测试方式)**

此方式用于将一个本地的 MP4 文件作为模拟直播源进行推流。

```bash
# 请将 <YOUR_SRS_IP> 和 <YOUR_STREAM_KEY> 替换为您的实际信息
ffmpeg -re -i /path/to/your/video.mp4 -c:v copy -c:a copy -f flv "rtmp://<YOUR_SRS_IP>:1935/live/<YOUR_STREAM_KEY>"
```

##### **场景 B: 推送无人值守的循环视频流 (24/7 直播)**

使用 `-stream_loop -1` 参数可以使视频文件无限循环播放。

```bash
# 无限循环播放 my_video.mp4
ffmpeg -re -stream_loop -1 -i /path/to/my_video.mp4 -c:v copy -c:a copy -f flv "rtmp://<YOUR_SRS_IP>:1935/live/<YOUR_STREAM_KEY>"
```

##### **场景 C: 推送实时摄像头画面 (真实直播)**

此方式用于捕获本机的摄像头和麦克风进行真实直播。**注意**: 输入设备名称因操作系统而异 (`avfoundation` for macOS, `dshow` for Windows, `video4linux2` for Linux)。

**macOS 示例:**

```bash
# -f avfoundation 指定输入格式
# -i "0:0" 表示使用第一个视频设备和第一个音频设备
ffmpeg -f avfoundation -i "0:0" -c:v libx264 -preset ultrafast -c:a aac -f flv "rtmp://<YOUR_SRS_IP>:1935/live/<YOUR_STREAM_KEY>"
```

**Windows 示例 (设备名称可能需要调整):**

```bash
# 需要先通过命令 `ffmpeg -list_devices true -f dshow -i dummy` 查找设备名称
ffmpeg -f dshow -i video="Integrated Camera":audio="Microphone Array" -c:v libx264 -preset ultrafast -c:a aac -f flv "rtmp://<YOUR_SRS_IP>:1935/live/<YOUR_STREAM_KEY>"
```

  - `image_url?: string`

- **成功响应（200/201）**：`success_response(data=Tab)`
- **错误码**
  - 400 / 1001：参数校验失败（content_type 逻辑不匹配等）
  - 403 / 3002：权限不足
  - 404 / 2004：房间不存在
  - 409 / 1009：tab_key 冲突（如实现了唯一约束/业务校验）
- **实现流程（摘要）**：见附录 `5.2.2 创建 Tab`，包含 room 存在性校验、content_type 规则校验、CRUD create + commit/refresh。

#### 2.4 更新 Tab
- **Endpoint**: `PATCH /api/v1/admin/rooms/{room_id}/tabs/{tab_id}`
- **权限**: `role ∈ {ADMIN, SUPERADMIN}`
- **请求体**：允许部分更新（title/content_type/text_content/image_url/sort_order）
- **成功响应**：`success_response(data=Tab)`
- **错误码**：403/404/400/500 同上（补充 tab 不存在）
- **实现流程**：按 V4 风格：权限校验 → room/tab 存在性 → 参数校验 → CRUD update

#### 2.5 删除 Tab
- **Endpoint**: `DELETE /api/v1/admin/rooms/{room_id}/tabs/{tab_id}`
- **权限**: `role ∈ {ADMIN, SUPERADMIN}`
- **成功响应**：`success_response(data={"deleted": true})` 或空 data
- **错误码**：403/404/500
- **实现流程**：权限校验 → tab 存在性 → CRUD delete → commit

### 3) 留言 API（观众 & 管理员）

#### 3.1 发送留言
- **Endpoint**: `POST /api/v1/rooms/{room_id}/messages`
- **权限**：已登录用户；角色来自 JWT（REGULAR/MODERATOR/ADMIN/SUPERADMIN）
- **请求体（最小）**
  - `content: string`（非空，长度限制按业务）
  - `session_id?: uuid`（可选；若留言与某场次绑定）
- **Message 字段扩展**：
  - `user_display_name: string | null` —— 用户展示名（昵称/用户名/邮箱快照），用于前端直接展示用户名。
- **成功响应（200/201）**：`success_response(data=Message)`，其中 `Message.user_display_name` 由后端填充。
- **错误码**
  - 400 / 1001：参数非法（空 content 等）
  - 404 / 2004：房间不存在
  - 401 / 3001：未登录/鉴权失败（按 V4 体系）
- **实现流程（摘要）**：
  1. **API 层**：从 JWT 解析得到的 `current_user` 中提取 `public_id`、`role`，并计算展示名：优先 `nickname`，其次 `username`，最后 `email`，得到 `user_display_name`。
  2. **Service 层**：`MessageService.create_message(..., user_display_name=user_display_name)` 在构造内部 DTO 时，将展示名写入 `live_room_messages.extra.user_display_name`，并按 V4/V6 规范执行业务校验和调用 CRUD。
  3. **CRUD 层**：按既有“学院派”事务模型将 `extra` 一并持久化到数据库，并将数据库异常转换为自定义异常上浮。
  4. **API 层**：在构造 `Message` DTO 时，从 ORM 对象的 `extra` 中读取 `user_display_name` 回填响应，最终通过 `success_response` 返回。
  5. **详细流程与异常映射**：见附录 `5.3.1 发送留言`（API→Service→CRUD→commit/refresh→返回）。

#### 3.2 获取留言列表（按直播间）
- **Endpoint**: `GET /api/v1/rooms/{room_id}/messages`
- **权限**：已登录（或按你系统策略允许匿名；若允许需明确）
- **Query（建议）**：分页参数 `limit/offset` 或 `page/page_size`
- **成功响应**：`success_response(data=[Message,...])`，其中每条 `Message` 对象均包含可选字段 `user_display_name: string | null`。
- **错误码**：404/401/500
- **实现流程（摘要）**：
  1. **Service 层**：保持原有分页查询逻辑不变，返回 ORM 模型列表 `messages`，其中 `extra` 字段可能包含 `user_display_name`。
  2. **API 层**：在将每条 ORM 记录映射为 `Message` DTO 时：
     - 若 `msg.extra` 为字典且包含 `user_display_name`，则填充/覆盖 DTO 的同名字段；
     - 否则保持 `user_display_name = null`，由前端按“匿名用户 / 脱敏 user_id”等兜底策略展示。
  3. **详细流程**：见附录 `5.3.2 获取留言列表（按直播间）`。

### 4) Import 创建 Session（新增，不影响旧路由）

#### 4.1 导入创建 session（带 playback_url）
- **Endpoint**: `POST /api/v1/rooms/{room_id}/sessions/import`
- **功能描述**：为"已存在回放的 session（从外部拷贝）"创建场次，并允许 **在创建时直接写入 playback_url**。
- **设计动机**：不污染普通创建语义；避免"先创建→改 finished→再 PATCH playback_url"的繁琐流程。
- **请求体（示例）**：
  ```json
  {
    "start_time": "2025-07-07T18:00:00Z",
    "end_time": "2025-07-07T19:00:00Z",
    "status": "finished",
    "playback_url": "https://mp2.dayilive.com/mp_clip/xxxx.m3u8"
  }
  ```

- **参数校验**：
  - `playback_url`：**必填**（Import 的意义就是携带回放）；URL 长度 <= 1024；仅校验格式，不做可播放性探测（避免引入外部依赖）
  - `status`：限制为 `finished`、`ready` 或 `live`（支持导入正在直播、已完成或就绪的场次）
  - `start_time`：**必填**（ISO 8601 格式）
  - `end_time`：建议必填（导入历史回放时通常明确）
  - （可选）其他字段：`title`、`description` 等，按 V4 session 字段口径

- **成功响应（200 OK）**：返回 session 详情，包含 playback_url（与 `GET /api/v1/sessions/{id}` 一致字段集合）
  ```json
  {
    "code": 200,
    "message": "success",
    "data": {
      "id": "session_uuid",
      "room_id": "room_uuid",
      "status": "finished",
      "start_time": "...",
      "end_time": "...",
      "video_id": null,
      "playback_url": "https://..."
    },
    "timestamp": "..."
  }
  ```

- **失败响应（示例）**：
  - 400 / 4xxx：参数校验失败（playback_url 缺失/过长、status 不在允许范围等）
  - 404 / 2004：room 不存在（HTTP 404 + 业务错误码）
  - 401 / 3xxx：未登录/鉴权失败
  - 403 / 3xxx：无权限（若导入仅限特定角色）

- **兼容性保证**：旧的 `POST /api/v1/rooms/{room_id}/sessions` 不变；前端只需在"普通创建/导入创建"上做分支选择。

- **实现流程（对齐 C 文档）**：
  1. 新增 Import Router（不改旧 Router）
  2. Service 复用现有 CRUD：创建 session 时直接写入 `playback_url`，并设置合适 status（避免"先 finished 再写回"的繁琐）
  3. 保持路由隔离，避免影响已有调用方

### 5) Search（模糊查询：按 room id / title）

> **策略**：优先“最小侵入”：扩展现有 list rooms 的 query 参数；如需更清晰可新增专用 search 路由（二选一，默认不破坏旧路由）。

#### 5.1 扩展现有 list rooms（推荐）
- **Endpoint**: `GET /api/v1/rooms`
- **新增 Query 参数**（可选，不影响旧调用）：
  - `q: string`（字符串，模糊匹配 room_id 或 title）
    - 若 `q` 看起来像 UUID：优先按 `id == q` 或 `id::text ILIKE %q%` 精确匹配
    - 同时支持 `title ILIKE %q%` 模糊匹配
    - 两种条件为 OR 关系，满足任一即可
- **兼容性**：旧调用不传 `q` 参数时，行为完全一致
- **分页返回**：保持 v4 的 `{total, page, size, items}` 格式
- **实现要点**：CRUD 层只加不改，SQL 使用 `ILIKE`（title）+ `id::text LIKE`（room_id）

#### 5.2（可选）新增专用 search 路由
- **Endpoint**: `GET /api/v1/rooms/search?q=...`
- **功能**：语义更清晰的搜索入口（可选实现）
- **Query 参数**：与 5.1 相同，支持 `q` 参数
- **响应**：与 list rooms 一致，避免多套返回结构
- **注意**：若路由稳定性要求极高，优先使用 5.1（仅可选参数扩展）

### 6) 批量导入（CSV/Excel：创建 room + session 并写入 playback_url）

#### 6.1 新增：批量导入接口
- **Endpoint**: `POST /api/v1/rooms/import/batch`
- **Content-Type**：`multipart/form-data`
- **上传文件**：`file`（.csv 或 .xlsx）
- **请求 Query 参数（可选）**：
  - `mode=dry_run|apply`（默认 `dry_run`，符合"安全与可回溯"原则）
    - `dry_run`：只校验不写入数据库，返回校验报告
    - `apply`：执行实际导入，写数据库并返回落库结果
  - `encoding_hint=utf-8-sig|utf-8|gbk`（可选；默认自动探测：utf-8-sig → utf-8 → gbk，与现有 CSV 容错逻辑一致）

- **文件字段契约（最小集合）**：
  - **必需列**：
    - `room_title`（直播间标题）
    - `playback_url`（回放地址）
  - **可选列**：
    - `room_id`（若外部指定；若不提供则系统生成 UUID）
    - `session_id`（同上）
    - `title`（session 标题，如没有则复用 room_title 或置空）
    - `start_time`, `end_time`（ISO 8601 格式）
    - `status`（默认 `finished`；导入历史回放最常见，也允许文件内覆盖为 `ready`/`finished`/`live`）
    - `description`（房间描述）
    - `cover_url`（封面URL）
  - **表头兼容说明（与微赞原始导出文件一致）**：
    - 实际业务中常见的源文件表头为中文，例如：`直播间ID, 标题, 直播间url, 创建时间, 开始时间, 结束时间, 播放url, 封面图片, ...`。
    - 批量导入模块在解析 CSV/Excel 时，应先对表头做**标准化与映射**，再还原到上述逻辑字段，其中推荐的最小映射关系为：
      - `标题` → `room_title`
      - `播放url`（以及历史模板中的 `播放url1`）→ `playback_url`
      - `封面图片` → `cover_url`
    - 换言之，**逻辑必需字段仍然是 `room_title` 与 `playback_url`**；无论是直接使用英文表头还是中文表头，经映射后仍需满足这两个字段齐全。若原始文件既不包含这些英文列，也不包含等价的中文列，应返回错误提示：`"CSV 文件缺少必需列：room_title, playback_url"`。

- **成功响应（200 OK）**：
  - `data` 中返回：
    - `total_rows`：总行数
    - `success_count`：成功导入数量
    - `failed_count`：失败数量
    - `items`：每行导入结果数组，每项包含：
      - `row_no`：行号（从 1 开始）
      - `room_id`：创建的房间 ID（成功时）
      - `session_id`：创建的场次 ID（成功时）
      - `status`：`"success"` 或 `"failed"`
      - `error`：失败原因（失败时）

- **失败响应（示例）**：
  - 400 / 4xxx：文件格式错误、必需列缺失、参数校验失败
  - 401 / 3xxx：未登录/鉴权失败
  - 403 / 3xxx：无权限（若导入仅限特定角色）

- **实现流程（Service 层）**：
  1. 标准化文本（trim、把多余空白折叠为单空格；不改变中英文内容）
  2. 对每一行：
     - 创建或复用 room：
       - 若提供 `room_id` 且存在：复用（权限校验必须做）
       - 否则创建新 room（使用现有创建逻辑生成 stream_key）
     - 创建 session（Import 语义）：
       - 直接走 **Service 的 import_create_session** 内部方法（复用逻辑）
       - `playback_url` 写入 `live_sessions.playback_url`
       - `status` 默认 `finished`（导入历史回放最常见）
     - 记录每行结果：success / fail + reason（用于导入报告）
  3. 若 `mode=dry_run`：不写 DB，只返回校验结果
  4. 若 `mode=apply`：写 DB，并返回落库结果

- **注意**：批量导入创建 room 时，需要一个**可预测且不冲突**的 `stream_key` 生成策略（例如 UUID 派生或随机），但这属于现有创建 room 的内部策略，不应破坏现有逻辑。

## 1\. 概述

本文档旨在为 LiveCore Service 增加“直播间 Tab 配置”和“直播间留言”两大功能。

## 2\. 需求背景

### 2.1 直播间 Tab 功能（管理员配置）

管理员在创建直播间后，可以为该直播间配置 0\~N 个 Tab（可选）。Tab 类型支持图文混合，用于展示「直播间简介」「医生简介」等。

## 3\. 数据流与模块说明（学院派架构）

本次新增功能的实现将严格遵循“学院派”分层架构：

1.  **API (router) 层**:
      * 负责接收 HTTP 请求，解析参数，并 `Depends` 注入 `db` 和 `current_user: Dict = Depends(get_current_user)`。
      * **必须**提前提取日志所需变量（如 `public_id = UUID(current_user.get("public_id"))`）。
      * **必须**使用 `try...except` 块调用 Service 层。
      * **必须**捕获 Service 层和 CRUD 层上浮的自定义异常（如 `RoomNotFoundException`, `DatabaseIntegrityException`），并将其转换为标准 `JSONResponse` 错误。
2.  **Service (业务) 层**:
      * 负责所有业务逻辑（如“检查用户角色是否允许发 URL”、“检查 Tab 管理权限”）。
      * **严禁**抛出 `HTTPException`，**必须**抛出自定义 Python 异常（如 `PermissionDeniedException`）。
      * **严禁**处理数据库事务（`db.commit()` / `db.rollback()`）。
3.  **CRUD (数据) 层**:
      * 负责所有数据库原子操作。
      * **必须**在函数内部处理事务（`try/except/finally`, `db.commit()`, `db.rollback()`）。
      * **必须**捕获 `IntegrityError` 等数据库错误，记录 `logger.error`，并 `raise` 自定义的数据库异常（如 `DatabaseIntegrityException`）。

### 7.2 异常处理流程

1.  **CRUD 层**:
    ```python
    # crud/messages.py
    async def create(db: AsyncSession, obj_in: dict, ...):
        # (安全规范) 提前提取日志变量
        public_id_log = obj_in.get("user_id")  # user_id 字段存储的是 public_id
        try:
            db_obj = Model(**obj_in)
            db.add(db_obj)
            await db.commit()
            await db.refresh(db_obj)
            return db_obj
        except IntegrityError as e:
            await db.rollback()
            logger.error(f"创建留言DB失败: public_id={public_id_log}, error={e}", exc_info=True)
            raise DatabaseIntegrityException("留言创建失败")
        except Exception as e:
            await db.rollback()
            logger.error(f"创建留言未知DB失败: public_id={public_id_log}, error={e}", exc_info=True)
            raise DatabaseOperationException("数据库操作失败")
    ```
2.  **Service 层**:
    ```python

        if user_role == 'REGULAR' and 'http' in content:
            # 严禁 HTTPExcepion，必须 raise 自定义异常

        # Service 层不捕获 CRUD 的异常，任其上浮
        db_message = await self.crud.create(db=self.db, ...)
        return db_message
    ```
3.  **API (router) 层**:
    ```python
    # routers/messages.py
    @router.post(...)
    async def create_message(..., current_user: Dict = Depends(get_current_user)):
        # (安全规范) 提前提取日志变量
        public_id_log = UUID(current_user.get("public_id"))
        role_log = current_user.get("role", "REGULAR").upper()
        
        try:
            service = MessageService(db)
            msg = await service.create_message(public_id=public_id_log, user_role=role_log, ...)
            return success_response(data=msg)
        
        # 捕获 Service 层的业务异常
        except InvalidParameterException as e:
            logger.warning(f"参数错误: public_id={public_id_log}, error={e.message}")
            return JSONResponse(status_code=400, content=error_response(code=e.code, message=e.message))
        
        # 捕获 Service 层的权限异常
        except PermissionDeniedException as e:
            logger.warning(f"权限不足: public_id={public_id_log}")
            return JSONResponse(status_code=403, content=error_response(code=3002, message="权限不足"))
            
        # 捕获 CRUD 层的数据库异常
        except DatabaseIntegrityException as e:
            logger.warning(f"数据冲突: public_id={public_id_log}, error={e.message}")
            return JSONResponse(status_code=400, content=error_response(code=4001, message="数据冲突或参数错误"))
            
        # 捕获所有其他异常
        except Exception as e:
            logger.error(f"未知错误: public_id={public_id_log}, error={e}", exc_info=True)
            return JSONResponse(status_code=500, content=error_response(code=1000, message="服务器内部错误"))
    ```

````md
## 1. 背景与问题定义（增量动机）

v4 已将 `playback_url` 作为 `live_sessions` 的持久化字段（`VARCHAR(1024) NULL`），并规定：
- **普通创建 session**：`playback_url` 默认应为 `NULL`  
- **后台任务**：在特定状态（如 ready 且 video_id 非空等）可以生成默认回放地址，但不得覆盖手动设置  
- **PATCH session**：在 ready/finished 状态允许修改 playback_url，且响应必须返回更新后的 playback_url（与 GET 一致）

当前你的痛点是：
- “从外部拷贝来的 live session”在创建时就已经有 `playback_url`，但你不得不先创建→改 finished→再 PATCH playback_url，操作繁琐且容易误用。
- 需要一个“导入/导出/检索”友好的增量能力：支持单个导入与批量导入；支持按 id/title 模糊搜索并快速定位。

> 结论：引入 **Import 类型创建** 是最小侵入且最符合原则的方案：不改变普通创建语义，仅新增一条导入路径专门处理“已存在回放”的 session。

### 2.1 与 v4 一致的规范（不再重复，仅强调一致性）
- **统一响应结构**：`{code, message, data, timestamp}`（v4 的通用响应规范）

- **日志/异常**：遵循 v4 的日志格式字段与异常分类（系统/业务/参数/权限）  
- **数据库访问**：保持既有 ORM/Session 管理方式，不引入与现有冲突的 DB 访问层

### 2.2 增量与最小修改原则（本增量的硬约束）
1) **旧路由不改**：现有 `POST /api/v1/rooms/{room_id}/sessions` 仍保持“普通创建”语义（playback_url 必须为 NULL）。  
2) **新增而非修改**：导入能力通过 **新增 Endpoint + 新 DTO** 实现；避免在旧 DTO 上加必填字段导致兼容性风险。  
3) **测试不破坏**：旧测试用例不改断言；新增测试覆盖导入/search/batch 导入。  
4) **可扩展与通用**：导入能力未来可支持更多字段（例如外部 source、external_id），但当前先最小落地：只解决 `playback_url`。

## 4. API 设计（新增，不影响旧路由）

> 以下新增 API 均遵循 v4 的统一响应结构与分页约定。

### 4.1 Import 创建 Session（新增 Endpoint）

#### 4.1.1 新增：导入创建 session
- **Endpoint**：`POST /api/v1/rooms/{room_id}/sessions/import`
- **功能描述**：为“已存在回放的 session（从外部拷贝）”创建场次，并允许 **在创建时直接写入 playback_url**。
- **设计动机**：不污染普通创建语义；避免你现在“先改 finished 再 PATCH”的繁琐流程。
- **请求体（示例）**：
```json
{
  "start_time": "2025-07-07T18:00:00Z",
  "end_time": "2025-07-07T19:00:00Z",
  "status": "finished",
  "playback_url": "https://mp2.dayilive.com/mp_clip/xxxx.m3u8"
}
````

* **参数校验**：

  * `playback_url` 必填（Import 的意义就是携带回放）
  * `status` 限制为：`finished`、`ready` 或 `live`
  * `start_time` 必填；`end_time` 建议必填（导入历史回放时通常明确）
  * `playback_url` 仅校验格式（URL 长度 <= 1024），不做可播放性探测（避免引入外部依赖）

* **成功响应（200 OK）**：返回 session 详情，包含 playback_url（与 `GET /api/v1/sessions/{id}` 一致字段集合）

```json
{
  "code": 200,
  "message": "success",
  "data": {
    "id": "session_uuid",
    "room_id": "room_uuid",
    "status": "finished",
    "start_time": "...",
    "end_time": "...",
    "video_id": null,
    "playback_url": "https://..."
  },
  "timestamp": "..."
}
```

* **失败响应（示例）**：

  * 参数校验失败：`4xxx`
  * room 不存在：`404`（HTTP） + `code=2004/4004`（按你现有规范映射）
  * 无权限：`401/403`（HTTP）+ `3xxx`

> **兼容性保证**：旧的 `POST /api/v1/rooms/{room_id}/sessions` 不变；前端只需在“普通创建/导入创建”上做分支选择。

### 4.2 Search（模糊查询：按 room id / title）

你希望类似截图：支持按 **直播间 id 或 title** 搜索，并分页展示列表。

#### 4.2.1 最小侵入方案（推荐）：扩展现有 list rooms 的 Query 参数（可选）

* **Endpoint**：仍使用 `GET /api/v1/rooms`
* **新增可选 Query 参数**：

  * `q`：字符串，模糊匹配（id 精确匹配 + title ILIKE）
  * 或拆分为：

    * `room_id`（精确）
    * `title_like`（模糊）
* **兼容性**：旧调用不传参数，行为完全一致。
* **分页返回**：保持 v4 的 `{total,page,size,items}` 格式。

**实现要点（Service/CRUD）**：

* 若 `q` 看起来像 UUID：优先按 `id == q` 或 `id::text ILIKE`（按你 DB 类型）
* 同时支持 `title ILIKE %q%`
* 返回 items 时字段保持现有 rooms list 输出字段集合

* `GET /api/v1/rooms/search?q=...`
* 仍然不改旧路由，但会新增一个新入口
* 缺点：前端与文档多一条路由；优点：语义更清晰

### 4.3 批量导入（CSV/Excel：创建 room + session 并写入 playback_url）

#### 4.3.1 新增：批量导入接口

* **Endpoint**：`POST /api/v1/import/sessions`
* **Content-Type**：`multipart/form-data`
* **上传文件**：`file`（.csv 或 .xlsx）
* **模式**：支持 dry-run（只校验不写入）与 apply（落库）

**请求 Query 参数（可选）**：

* `mode=dry_run|apply`（默认 dry_run，符合“安全与可回溯”）
* `encoding_hint=utf-8-sig|utf-8|gbk`（可选；默认自动探测：utf-8-sig → utf-8 → gbk，与你现有 CSV 容错逻辑一致）

#### 4.3.2 文件字段契约（最小集合）

你的文件包含：直播间 id、title、playback_url 等关键信息。为保证通用性，定义最小必需列：

* **必需列**：

  * `room_title`（直播间标题）
  * `session_title`（可选，如没有则复用 room_title 或置空；看你现有 session 是否有 title 字段）
  * `playback_url`（回放地址）
* **可选列**：

  * `room_id`（若外部指定。若不提供则系统生成 UUID）
  * `session_id`（同上）
  * `start_time`, `end_time`, `status`（若不提供，按导入默认策略）

> 注意：你 v4 的 DDL 里 `live_rooms.title` 必填、`stream_key` 必填且唯一。批量导入创建 room 时，需要一个**可预测且不冲突**的 stream_key 生成策略（例如 UUID 派生或随机），但这属于现有创建 room 的内部策略，不应破坏现有逻辑。

#### 4.3.3 导入业务流程（Service 层）

对每一行：

1. 标准化文本（trim、把多余空白折叠为单空格；不改变中英文内容）
2. 创建或复用 room：

   * 若提供 `room_id` 且存在：复用（权限校验必须做）
   * 否则创建新 room（使用你现有创建逻辑生成 stream_key）
3. 创建 session（Import 语义）：

   * 直接走 **Service 的 import_create_session** 内部方法（不要复制逻辑）
   * playback_url 写入 live_sessions.playback_url
   * status 默认 `finished`（导入历史回放最常见），也允许文件内覆盖为 ready/finished/live
4. 记录每行结果：success / fail + reason（用于导入报告）

#### 4.3.4 响应（200 OK）

* `data` 中返回：

  * `total_rows`
  * `success_count`
  * `failed_count`
  * `items`（每行导入结果：row_no、room_id、session_id、status、error）
* 若 mode=dry_run：不写 DB，只返回校验结果
* 若 mode=apply：写 DB，并返回落库结果



#### 4.3.4 V6 批量导入幂等与去重扩展

> 本小节在不改变 4.3.1 ~ 4.3.3 既有接口与字段语义的前提下，为批量导入补充 **room 维度去重 + session 维度幂等** 能力。V5 及更早版本的行为可以视为“未开启幂等扩展”的情况。

##### 4.3.4.1 Room 维度去重（external_room_id）

在 4.3.3 中，“创建或复用 room” 的逻辑在 V6 中扩展为：

1. **若提供 `room_id` 且存在且属于当前用户**：仍按既有逻辑复用该房间（包含权限校验）。
2. **否则，若提供 `external_room_id`**：
   - 通过 `(user_id, external_room_id)` 在 `live_rooms` 中查找；
   - 查到则直接复用该 room；
   - 查不到则创建新 room，并将该 external_room_id 写入 `live_rooms.external_room_id` 字段。
3. **否则**：按 4.3.3 原有规则创建新 room（不依赖 external_room_id）。

其中 `external_room_id` 与导入文件的映射关系建议为：

- `external_room_id` ← CSV/Excel 表头中的 `external_room_id` / `直播间id` / `直播间ID` 等别名（由表头映射函数统一处理）。

**表头别名映射详细规则**：

在 `BatchImportService._apply_header_aliases` 方法中，需要将以下表头（不区分大小写，去除首尾空格后匹配）统一映射到逻辑字段 `external_room_id`：

- `"external_room_id"`
- `"直播间id"`
- `"直播间ID"`

映射后的行数据中应统一使用 `row["external_room_id"]`，后续 `_get_or_create_room_for_row` 可直接使用该字段进行房间幂等判断。

##### 4.3.4.2 Session 维度幂等（playback_url_hash）

在 4.3.3 的“创建 session（Import 语义）”步骤中，V6 引入 `playback_url_hash` 辅助字段与 Service 层幂等逻辑：

**URL 规范化与哈希计算函数**：

```python
def normalize_playback_url(url: str) -> str:
    """
    规范化 playback_url，至少去除首尾空白。
    用于确保相同 URL 的不同表示形式（如带空格）能够被正确识别为同一 URL。
    """
    return url.strip()

def calc_playback_url_hash(url: str) -> str:
    """
    统一的 playback_url 哈希计算函数。
    内部会先调用 normalize_playback_url，避免调用方忘记先规范化。
    推荐使用 SHA-256 十六进制字符串（长度为 64 字符）。
    """
    from hashlib import sha256
    normalized = normalize_playback_url(url)
    return sha256(normalized.encode("utf-8")).hexdigest()
```

**幂等逻辑流程**：

1. 在 `SessionImportService.import_create_session(room_id, payload, user_id)` 内部：
   - 对 `payload.playback_url` 调用 `normalize_playback_url(url)` 做标准化（至少去除首尾空白）；
   - 调用 `calc_playback_url_hash(url)` 计算哈希，并写入 `playback_url_hash` 字段；
   - 使用 `(room_id, playback_url_hash)` 在 `live_sessions` 中查询是否已有记录。
2. 若存在记录：
   - 视为 **幂等命中**，直接返回已有 `LiveSession`；
   - 不再创建新记录。
3. 若不存在记录：
   - 创建新的 `LiveSession`，写入 `playback_url`（规范化后的值）与 `playback_url_hash`；
   - 如发生唯一约束冲突（并发场景），捕获 `IntegrityError` 异常后再查一次并返回已有记录，确保最终只有一条记录。

**并发场景处理**：

在并发导入同一 `room_id + playback_url` 的场景下，可能出现多个请求同时执行"查询不存在 → 创建新记录"的情况。此时：

1. 数据库的唯一索引 `uq_live_sessions_room_playback_hash` 会在数据库层面保证最终只存在一条记录。
2. Service 层应捕获 `IntegrityError` 异常，将其转换为"命中已有记录"的行为：
   - 捕获异常后，再次查询 `(room_id, playback_url_hash)`；
   - 返回查询到的已有记录（此时必然存在，因为另一个并发请求已创建）；
   - 记录日志，但不抛出异常，确保导入流程继续执行。
3. CRUD 层负责捕获 `IntegrityError` 并转换为 `DatabaseIntegrityException`，Service 层捕获后按上述逻辑处理。

##### 4.3.4.3 导入结果结构中的幂等标记

在保持原有导入结果结构的前提下，V6 为每一行的导入结果增加以下**可选字段**（旧客户端可忽略）：

- `skipped: bool`：是否因为幂等命中而“逻辑成功但未真正新建记录”；
- `skip_reason: Optional[str]`：目前主要值为 `"duplicate_session_by_playback_url"`，未来可扩展为更细粒度的枚举。

其中：

- `status` 字段仍然只有 `"success"` / `"failed"` 两种；
- `success_count` 仍然统计所有 `status="success"` 的行（包括 `skipped=true` 的幂等行），以保持与 V5 前后口径一致。

##### 4.3.4.4 覆盖的幂等场景

通过 4.3.4.1 与 4.3.4.2 的组合，V6 能够覆盖以下两类常见重复场景：

- **同一 CSV 文件被重复导入多次**：
  - external_room_id 使得同一外部直播间 ID 只对应一个 room；
  - playback_url_hash 使得同一 room 下相同 playback_url 只对应一个 session。
- **不同 CSV 文件包含相同行（同一外部直播间 ID + 同一 playback_url）**：
  - 依旧会命中同一个 room + session，不会产生重复记录。

##### 4.3.4.5 playback_url_hash 维护范围（Service 层强约束）

在实现层面，除了在 Import Session / Batch Import 场景中写入 `playback_url_hash` 外，还必须满足以下统一约束：

**所有会修改 `playback_url` 的 Service 方法**（包括但不限于）：
- `SessionImportService.import_create_session`；
- `BatchImportService` 中调用 Import Session 的路径；
- `SessionService.update_scheduled_session_info` 中允许更新回放地址的分支；
- `SessionService.auto_generate_playback_url` 等后台生成回放地址的方法；

**都必须在更新 `playback_url` 时，同步调用**：

```python
playback_url_hash = calc_playback_url_hash(new_url)
```

将计算结果写入 `live_sessions.playback_url_hash` 字段。

> **重要约束**：凡是更新 `playback_url` 的 Service 方法，都必须同步维护 `playback_url_hash`。任何只改 URL 而不更新 hash 的实现都视为不符合本设计约定，可能导致幂等逻辑失效。

```python
class SessionImportService:
    async def import_create_session(
        self,
        room_id: UUID,
        public_id: UUID,
        session_in: dict,
    ) -> Tuple[LiveSession, bool]:
        """
        导入创建会话（支持幂等）。
        
        返回: (session, idempotent_hit)
        - idempotent_hit=True: 表示命中已有记录，未创建新记录
        - idempotent_hit=False: 表示创建了新记录
        """
        # 1. 房间存在性 + 权限检查（保持 V5 行为）
        room = await crud_room.get(self.db, room_id=room_id, user_id=public_id)
        if not room:
            raise RoomNotFoundException(f"房间不存在或无权限: {room_id}")
        
        if room.user_id != public_id:
            raise PermissionDeniedException("无权在该房间导入会话")
        
        # 2. 参数校验
        status = session_in.get("status")
        if status not in ["finished", "ready", "live"]:
            raise InvalidParameterException("Import Session 的 status 只能为 'finished'、'ready' 或 'live'")
        
        playback_url = session_in.get("playback_url")
        if not playback_url:
            raise InvalidParameterException("Import Session 必须提供 playback_url")
        
        # 3. URL 规范化与哈希计算
        normalized_url = normalize_playback_url(playback_url)
        playback_hash = calc_playback_url_hash(playback_url)
        
        # 4. 幂等查询
        existing = await crud_session.get_by_room_and_playback_hash(
            db=self.db,
            room_id=room_id,
            playback_url_hash=playback_hash,
        )
        if existing:
            return existing, True  # True 表示幂等命中
        
        # 5. 创建会话
        try:
            session_data = {
                "id": uuid.uuid4(),
                "room_id": room_id,
                "status": status,
                "start_time": session_in.get("start_time"),
                "end_time": session_in.get("end_time"),
                "playback_url": normalized_url,
                "playback_url_hash": playback_hash,
            }
            db_session = await crud_session.create_with_stats(
                self.db, 
                obj_in=session_data,
                user_id=public_id
            )
            return db_session, False
        except DatabaseIntegrityException as e:
            # 并发场景：唯一约束冲突，再次查询并返回已有记录
            existing = await crud_session.get_by_room_and_playback_hash(
                db=self.db,
                room_id=room_id,
                playback_url_hash=playback_hash,
            )
            if existing:
                return existing, True
            raise  # 如果仍然查不到，重新抛出异常
```

```python
class BatchImportService:
    async def _get_or_create_room_for_row(
        self, 
        public_id: UUID, 
        row: Dict
    ) -> LiveRoom:
        """
        根据行数据获取或创建房间（支持 external_room_id 幂等）。
        """
        room_id_str = row.get("room_id")
        external_room_id = row.get("external_room_id")
        
        # a) 有 room_id：按 V5 行为复用
        if room_id_str:
            room_id = UUID(room_id_str)
            room = await crud_room.get(
                self.db, 
                room_id=room_id, 
                user_id=public_id
            )
            if not room:
                raise InvalidParameterException("房间不存在或无权限")
            return room
        
        # b) 有 external_room_id：按(user_id, external_room_id) 幂等
        if external_room_id:
            room = await crud_room.get_by_external_id(
                db=self.db,
                user_id=public_id,
                external_room_id=external_room_id,
            )
            if room:
                return room
        
        # c) 否则创建新房间
        room_data = {
            "title": row.get("room_title", "导入的房间"),
            "description": row.get("description"),
            "cover_url": row.get("cover_url"),
            # ... 其他字段
        }
        room = await crud_room.create(
            self.db, 
            obj_in=room_data, 
            user_id=public_id
        )
        
        # 如果提供了 external_room_id，写入该字段
        if external_room_id:
            room.external_room_id = external_room_id
            await self.db.commit()
            await self.db.refresh(room)
        
        return room
```

**BatchImportService._process_row**：

```python
class BatchImportService:
    async def _process_row(
        self,
        public_id: UUID,
        row_no: int,
        row: Dict,
        mode: str = "apply"
    ) -> Dict:
        """
        处理单行导入数据。
        
        返回: {
            "row_no": int,
            "room_id": str | None,
            "session_id": str | None,
            "status": "success" | "failed",
            "error": str | None,
            "skipped": bool,
            "skip_reason": str | None,
        }
        """
        try:
            # 1. 字段校验
            if not row.get("room_title") or not row.get("playback_url"):
                return {
                    "row_no": row_no,
                    "status": "failed",
                    "error": "缺少必需字段: room_title 或 playback_url",
                    "skipped": False,
                    "skip_reason": None,
                }
            
            # 2. 获取或创建房间
            room = await self._get_or_create_room_for_row(public_id, row)
            
            # 3. 处理会话（仅 apply 模式）
            if mode == "apply":
                session_in = {
                    "status": row.get("status", "finished"),
                    "start_time": self._parse_time(row.get("start_time")),
                    "end_time": self._parse_time(row.get("end_time")),
                    "playback_url": row["playback_url"],
                }
                session, idempotent_hit = await SessionImportService(
                    self.db
                ).import_create_session(
                    room_id=room.id,
                    public_id=public_id,
                    session_in=session_in,
                )
                return {
                    "row_no": row_no,
                    "room_id": str(room.id),
                    "session_id": str(session.id),
                    "status": "success",
                    "error": None,
                    "skipped": idempotent_hit,
                    "skip_reason": "duplicate_session_by_playback_url" if idempotent_hit else None,
                }
            else:
                # dry_run：仅做字段校验，不写库
                return {
                    "row_no": row_no,
                    "status": "success",
                    "error": None,
                    "skipped": False,
                    "skip_reason": None,
                }
        except Exception as e:
            return {
                "row_no": row_no,
                "status": "failed",
                "error": str(e),
                "skipped": False,
                "skip_reason": None,
            }
```

   - 任何唯一索引冲突异常被正确捕获并转化为“命中已有记录”的行为，而不会导致整体导入失败。

## 7. 你提到的“操作繁琐”的直接解决方案（最小改动达成）

你现在必须“先把 status 改 finished 保存，再更新 playback_url 才能更新”的原因，本质是：

* 你把“允许修改 playback_url”的规则写在了 `PATCH` 的业务约束里（只允许 ready/finished），而创建时走的是“普通创建”语义（playback_url 必须 NULL）。

引入 **Import 创建 session** 后：

* 外部拷贝来的 session 直接走 `POST /rooms/{room_id}/sessions/import`，请求体里包含 status=finished 与 playback_url
* 你不需要执行额外的 PATCH，流程减少为一步
* 普通创建仍保持 playback_url=NULL 的规则不变（与 v4 一致）
