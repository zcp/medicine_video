# ==================== 权限守卫函数 ====================

def _check_admin_permission(
    self,
    user_role: Optional[str]
) -> None:
    """
    检查管理员权限

    权限判断流程：
        1. Admin（role in ['ADMIN', 'SUPERADMIN']）→ 通过
        2. 其他角色 → 403（权限不足）

    Args:
        user_role: 当前用户的角色

    Raises:
        PermissionDeniedException(403): 权限不足
    """
    if user_role not in ['ADMIN', 'SUPERADMIN']:
        raise PermissionDeniedException("需要管理员权限")

def _check_write_permission(
    self,
    resource_user_id: UUID,
    current_user_id: UUID,
    user_role: Optional[str]
) -> None:
    """
    检查写入权限（资源所有权）

    权限判断流程：
        1. Admin（role in ['ADMIN', 'SUPERADMIN']）→ 上帝视角通过
        2. Owner（resource_user_id == current_user_id）→ 通过
        3. 其他情况 → 403（权限不足）

    Args:
        resource_user_id: 资源所属用户的ID
        current_user_id: 当前用户的ID
        user_role: 当前用户的角色

    Raises:
        PermissionDeniedException(403): 权限不足
    """
    # 管理员：上帝视角通过
    if user_role in ['ADMIN', 'SUPERADMIN']:
        return
    
    # 资源创建者：通过
    if resource_user_id == current_user_id:
        return
    
    # 其他用户：拒绝
    raise PermissionDeniedException("您没有权限操作该资源")
```

# ✅ 更新后
async def mark_notification_as_read(
    self,
    user_id: UUID,
    user_role: str,
    notification_id: UUID
) -> None:
    """将单条通知标记为已读"""
    
    # 提取日志变量
    user_id_log = user_id
    notification_id_log = notification_id
    
    try:
        # 1. 查询通知
        notification = await get_notification_by_id(self.db, notification_id, user_id)
        
        if notification is None:
            logger.warning(
                f"通知不存在或不属于该用户: notification_id={notification_id_log}, "
                f"user_id={user_id_log}"
            )
            raise NotFoundException("通知不存在")
        
        # 2. 幂等性检查：如果 is_read 已为 True，直接返回
        if notification.is_read:
            logger.info(
                f"通知已为已读状态,直接返回: notification_id={notification_id_log}"
            )
            return
        
        # 3. 权限检查（使用权限守卫函数）
        self._check_user_preferences_ownership(notification.user_id, user_id, user_role)
        
        # 4. 更新状态
        notification.is_read = True
        await self.db.commit()
        
        logger.info(
            f"标记通知为已读成功: notification_id={notification_id_log}, "
            f"user_id={user_id_log}"
        )
        
    except NotFoundException:
        raise
    except Exception as e:
        logger.error(
            f"标记通知为已读失败: notification_id={notification_id_log}, "
            f"user_id={user_id_log}, error={str(e)}",
            exc_info=True
        )
        raise
```
