## 二、数据库表设计

### 2.1 直播服务数据库

#### 2.1.1 直播流主表（live_stream）
```sql
CREATE TABLE live_stream (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id INT NOT NULL,
    title VARCHAR(100) NOT NULL,
    description TEXT,
    cover_url VARCHAR(255),
    stream_key VARCHAR(255) NOT NULL,
    storage_type VARCHAR(20) NOT NULL,  -- local, s3, etc.
    stream_path VARCHAR(255),
    storage_bucket VARCHAR(100),        -- 存储桶（如果使用对象存储）
    status VARCHAR(20) DEFAULT 'created',
    start_time TIMESTAMP,
    end_time TIMESTAMP,
    is_private BOOLEAN DEFAULT false,
    is_recorded BOOLEAN DEFAULT true,
    category_id INT,                    -- 直播分类
    sub_category_id INT,                -- 直播子分类
    thumbnail_url VARCHAR(255),
    scheduled_time TIMESTAMP,
    permission_level INT DEFAULT 0,     -- 权限级别
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.1.2 直播分类表（stream_categories）
```sql
CREATE TABLE stream_categories (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(50) NOT NULL,
    parent_id INT,
    sort_order INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.1.3 直播标签表（stream_tags）
```sql
CREATE TABLE stream_tags (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(50) NOT NULL UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.1.4 直播-标签关联表（stream_tag_relations）
```sql
CREATE TABLE stream_tag_relations (
    stream_id INT REFERENCES live_stream(id),
    tag_id INT REFERENCES stream_tags(id),
    PRIMARY KEY (stream_id, tag_id)
);
```

#### 2.1.5 直播统计表（stream_statistics）
```sql
CREATE TABLE stream_statistics (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    stream_id INT REFERENCES live_stream(id),
    total_viewer_count INT DEFAULT 0,
    peak_viewer_count INT DEFAULT 0,
    total_like_count INT DEFAULT 0,
    total_share_count INT DEFAULT 0,
    total_comment_count INT DEFAULT 0,
    last_view_time TIMESTAMP,             -- 最近观看时间
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.1.6 直播权限表（stream_permissions）
```sql
CREATE TABLE stream_permissions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id INT NOT NULL,
    permission_type VARCHAR(20) NOT NULL, -- create, read, update, delete
    is_public BOOLEAN DEFAULT false,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.1.7 直播权限规则表（stream_permission_rules）
```sql
CREATE TABLE stream_permission_rules (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    rule_type VARCHAR(50) NOT NULL,     -- user, role, vip, etc.
    rule_value TEXT NOT NULL,
    permission_type VARCHAR(20) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

### 2.2 回放服务数据库

#### 2.2.1 回放记录主表（playback_record）
```sql
CREATE TABLE playback_record (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    stream_id INT NOT NULL,
    user_id INT NOT NULL,
    title VARCHAR(100) NOT NULL,
    description TEXT,
    video_path VARCHAR(255) NOT NULL,
    duration INT,
    file_size BIGINT,
    resolution VARCHAR(50),
    format VARCHAR(20),
    storage_type VARCHAR(20),
    status VARCHAR(20) DEFAULT 'processing',
    is_public BOOLEAN DEFAULT true,
    permission_level INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.2.2 回放分片表（playback_segments）
```sql
CREATE TABLE playback_segments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    playback_id INT REFERENCES playback_record(id),
    segment_path VARCHAR(255) NOT NULL,
    segment_index INT NOT NULL,
    duration INT NOT NULL,
    file_size BIGINT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.2.3 回放质量表（playback_quality）
```sql
CREATE TABLE playback_quality (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    playback_id INT REFERENCES playback_record(id),
    bitrate INT,
    fps INT,
    audio_bitrate INT,
    audio_sample_rate INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.2.4 回放统计表（playback_stats）
```sql
CREATE TABLE playback_stats (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    playback_id INT REFERENCES playback_record(id),
    view_count INT DEFAULT 0,
    like_count INT DEFAULT 0,
    comment_count INT DEFAULT 0,
    share_count INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

CREATE TABLE stream_quality_metrics (

CREATE TABLE system_resource_metrics (

### 2.4 配置及存储服务数据库

#### 2.4.1 系统配置表（system_config）
```sql
CREATE TABLE system_config (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    config_key VARCHAR(100) NOT NULL,
    config_value TEXT,
    description TEXT,
    environment VARCHAR(20) NOT NULL,   -- dev, test, prod
    version INT NOT NULL,
    is_public BOOLEAN DEFAULT false,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(config_key, environment, version)
);
```

#### 2.4.2 配置变更历史表（config_change_history）
```sql
CREATE TABLE config_change_history (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    config_id INT REFERENCES system_config(id),
    old_value TEXT,
    new_value TEXT,
    changed_by INT NOT NULL,
    change_reason TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.4.3 存储类型表（storage_types）
```sql
CREATE TABLE storage_types (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    type_name VARCHAR(20) NOT NULL UNIQUE, -- s3, oss, local, etc.
    description TEXT,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.4.4 存储配置表（storage_configs）
```sql
CREATE TABLE storage_configs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    storage_type_id INT REFERENCES storage_types(id),
    environment VARCHAR(20) NOT NULL,   -- dev, test, prod
    version INT NOT NULL,
    config_data JSONB NOT NULL,         -- 存储所有配置项
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(storage_type_id, environment, version)
);
```

### 2.5 媒体资源服务数据库

#### 2.5.1 视频资源主表（video_resources）
```sql
CREATE TABLE video_resources (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    video_id VARCHAR(50) NOT NULL UNIQUE,  -- 遵循视频ID命名规范 {room_id}_{uuid4后缀}
    room_id VARCHAR(20) NOT NULL,          -- 直播间ID
    title VARCHAR(100) NOT NULL,
    description TEXT,
    storage_path VARCHAR(255) NOT NULL,    -- 遵循存储目录结构 /media/video_{video_id}/
    status VARCHAR(20) DEFAULT 'created',
    is_public BOOLEAN DEFAULT true,
    permission_level INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.5.2 视频文件表（video_files）
```sql
CREATE TABLE video_files (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    video_id VARCHAR(50) NOT NULL REFERENCES video_resources(video_id),
    file_name VARCHAR(255) NOT NULL,       -- 遵循文件命名规范
    file_type VARCHAR(20) NOT NULL,        -- raw, hls, mp4, cover, thumbnail
    operation_type VARCHAR(20) NOT NULL,   -- upload, transcoded
    file_path VARCHAR(255) NOT NULL,       -- 完整存储路径
    file_size BIGINT NOT NULL,
    resolution VARCHAR(50),                -- 视频分辨率
    bitrate INT,                          -- 视频码率
    format VARCHAR(20),                   -- 文件格式
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.5.3 视频分片表（video_segments）
```sql
CREATE TABLE video_segments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    video_id VARCHAR(50) NOT NULL REFERENCES video_resources(video_id),
    segment_name VARCHAR(255) NOT NULL,    -- 遵循文件命名规范
    segment_index INT NOT NULL,            -- 分片序号
    segment_path VARCHAR(255) NOT NULL,    -- 完整存储路径
    duration INT NOT NULL,                 -- 分片时长(秒)
    file_size BIGINT NOT NULL,
    status VARCHAR(20) DEFAULT 'created',  -- created, processing, completed, failed
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2.5.4 视频元数据表（video_metadata）
```sql
CREATE TABLE video_metadata (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    video_id VARCHAR(50) NOT NULL REFERENCES video_resources(video_id),
    metadata_key VARCHAR(50) NOT NULL,
    metadata_value TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(video_id, metadata_key)
);
```

#### 2.5.5 视频访问记录表（video_access_logs）
```sql
CREATE TABLE video_access_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    video_id VARCHAR(50) NOT NULL REFERENCES video_resources(video_id),
    user_id INT NOT NULL,
    access_type VARCHAR(20) NOT NULL,      -- view, download, share
    file_type VARCHAR(20) NOT NULL,        -- raw, hls, mp4, cover, thumbnail
    ip_address VARCHAR(50),
    user_agent TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 创建索引
CREATE INDEX idx_video_resources_video_id ON video_resources(video_id);
CREATE INDEX idx_video_files_video_id ON video_files(video_id);
CREATE INDEX idx_video_segments_video_id ON video_segments(video_id);
CREATE INDEX idx_video_metadata_video_id ON video_metadata(video_id);
CREATE INDEX idx_video_access_logs_video_id ON video_access_logs(video_id);
```
