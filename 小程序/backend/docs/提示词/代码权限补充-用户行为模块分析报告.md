```python
def _check_user_behavior_ownership(
    self,
    resource: Any,
    user_id: UUID,
    user_role: str
) -> None:
    """
    用户行为所有权校验（通用）
    
    权限判断流程：
        1. Admin（role in ['ADMIN', 'SUPERADMIN']）→ 上帝视角通过
        2. Owner（resource.user_id == user_id）→ 通过
        3. 其他情况 → 403（权限不足）
    
    Raises:
        PermissionDeniedException(403): 权限不足
    """
    # 管理员：上帝视角通过
    if user_role in ['ADMIN', 'SUPERADMIN']:
        return
    
    # 资源创建者：通过
    if hasattr(resource, 'user_id') and resource.user_id == user_id:
        return
    
    # 其他用户：拒绝
    raise PermissionDeniedException(
        "您没有权限访问其他用户的偏好设置"
    )
```
