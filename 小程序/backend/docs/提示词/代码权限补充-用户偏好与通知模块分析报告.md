async def get_all_users_preferences(

    获取所有用户偏好（管理员专属）

        skip: 跳过记录数
        limit: 每页记录数

    user_preferences_list = result.scalars().all()
    
    return user_preferences_list, total

    获取所有通知（管理员专属，支持按用户筛选）
    
    Args:
        db: 数据库会话
        skip: 跳过记录数
        limit: 每页记录数
        user_id: 按用户筛选（可选）
        is_read: 已读/未读筛选（可选）
    
    Returns:
        (List[Notification], total): 通知列表和总数
    """
    # 构建基础查询
    base_query = select(Notification)
    
    # 应用业务筛选和权限过滤
    if is_read is not None:
        base_query = base_query.where(Notification.is_read == is_read)
    
    if user_id is not None:
        # 按用户筛选
        base_query = base_query.where(Notification.user_id == user_id)
    
    # 获取总数
    count_query = select(func.count()).select_from(base_query)
    total_result = await db.execute(count_query)
    total = total_result.scalar()
    
    # 获取数据列表
    query = base_query.order_by(Notification.created_at.desc()).offset(skip).limit(limit)
    result = await db.execute(query)
    notifications_list = result.scalars().all()
    
    return notifications_list, total
