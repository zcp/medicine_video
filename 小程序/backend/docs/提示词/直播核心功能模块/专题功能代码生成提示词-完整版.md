**权限验证示例**:
```python
async def update_topic(self, topic_id: UUID, topic_in: TopicUpdate, user_id: UUID) -> Topic:
    """更新专题（仅创建者可操作）"""
    # 1. 查询专题
    topic = await crud_topic.get_topic(self.db, topic_id)
    if not topic:
        logger.error(f"专题不存在: topic_id={topic_id}")
        raise TopicNotFoundException(str(topic_id))
    
    # 2. 权限验证
    if topic.user_id != user_id:
        logger.error(f"权限验证失败: user_id={user_id}, topic_owner={topic.user_id}")
        raise TopicPermissionDeniedException("修改专题")
    
    # 3. 执行更新
    updated_topic = await crud_topic.update_topic(self.db, topic_id, topic_in)
    
    # 4. 日志记录
    logger.info(f"专题更新成功: user_id={user_id}, topic_id={topic_id}, fields={list(topic_in.dict(exclude_unset=True).keys())}")
    
    return updated_topic
```

**端点实现模板**:
```python
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from typing import Dict, Any

from app.core.deps import get_db, get_current_user
from app.schemas.topic import TopicCreate, TopicUpdate, TopicResponse
from app.services.topic_service import TopicService
from app.api.v1.responses import success_response

router = APIRouter()

@router.post("/topics", response_model=Dict[str, Any])
async def create_topic(
    topic_in: TopicCreate,
    current_user: dict = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
) -> Dict[str, Any]:
    """
    创建专题
    
    遵循：
    - 核心文档 Section 4.1: 统一响应结构
    - 核心文档 Section 9: JWT认证
    - 专题文档 Section 8.X.2.1: 创建专题API
    """
    try:
        # 1. 提取user_id
        user_id = UUID(current_user["user_id"])
        
        # 2. 调用Service层
        service = TopicService(db)
        topic = await service.create_topic(topic_in, user_id)
        
        # 3. 返回统一响应
        return success_response(data=TopicResponse.from_orm(topic))
        
    except Exception as e:
        logger.error(f"创建专题失败: user_id={user_id}, error={str(e)}")
        raise
```
