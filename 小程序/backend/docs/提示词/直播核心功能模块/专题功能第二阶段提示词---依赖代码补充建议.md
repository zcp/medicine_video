## ⚠️ 不附带代码的风险

### 1. **Import语句错误**
```python
# AI可能生成错误的import
from app.models.live_core import Topic  # ❌ Topic在topic.py中，不在live_core.py
from app.schemas.topic import TopicCreate  # ❌ 可能不知道确切的Schema名称
```

### 2. **字段访问错误**
```python
# AI可能访问不存在的字段
if topic.owner_id != user_id:  # ❌ 字段名是user_id，不是owner_id
    raise PermissionError()
```

### 3. **方法调用错误**
```python
# AI可能使用错误的方法签名
topic = await crud_topic.get(topic_id)  # ❌ 缺少db参数
# 正确：topic = await crud_topic.get(self.db, topic_id)
```

### 4. **类型注解错误**
```python
# AI可能使用错误的类型
async def update_topic(self, topic_id: UUID, data: dict) -> dict:  # ❌
# 正确：async def update_topic(self, topic_id: UUID, obj_in: TopicUpdate) -> Topic:
```

### 5. **Schema使用错误**
```python
# AI可能不知道有哪些Schema可用
@router.post("/")
async def create_topic(data: dict):  # ❌ 应该使用TopicCreate
    pass
```

"""
LiveCore Service - Topic Models

专题聚合功能的SQLAlchemy ORM模型定义
"""

import uuid
from datetime import datetime
from typing import Optional, List
import enum
from sqlalchemy import Column, String, Text, Integer, TIMESTAMP, ForeignKey, Enum as SAEnum
from sqlalchemy.dialects.postgresql import UUID
from sqlalchemy.orm import relationship
from sqlalchemy.sql import func

from app.database import Base


class TopicStatus(str, enum.Enum):
    """专题状态枚举"""
    DRAFT = "draft"
    PUBLISHED = "published"
    ARCHIVED = "archived"


class Topic(Base):
    """
    专题模型 - 对应 topics 表
    顶层实体，用于聚合多个分类和直播间
    """
    __tablename__ = "topics"
    
    # 主键 - UUID类型，应用层生成
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    
    # 用户ID - 通过JWT Token验证
    user_id = Column(UUID(as_uuid=True), nullable=False)
    
    # 基本信息
    title = Column(String(100), nullable=False)
    description = Column(Text, nullable=True)
    banner_url = Column(String(255), nullable=True)
    
    # 状态
    status = Column(
        SAEnum(TopicStatus, values_callable=lambda obj: [e.value for e in obj]),
        nullable=False,
        server_default="draft"
    )
    
    # 时间戳
    created_at = Column(TIMESTAMP(timezone=True), nullable=False, server_default=func.now())
    updated_at = Column(TIMESTAMP(timezone=True), nullable=False, server_default=func.now(), onupdate=func.now())
    
    # 关系定义
    categories = relationship("TopicCategory", back_populates="topic", cascade="all, delete-orphan")


class TopicCategory(Base):
    """
    专题分类模型 - 对应 topic_categories 表
    中层实体，用于组织直播间
    """
    __tablename__ = "topic_categories"
    
    # 主键
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    
    # 外键 - 专题ID
    topic_id = Column(UUID(as_uuid=True), ForeignKey("topics.id", ondelete="CASCADE"), nullable=False)
    
    # 基本信息
    name = Column(String(50), nullable=False)
    sort_order = Column(Integer, default=0)
    
    # 时间戳
    created_at = Column(TIMESTAMP(timezone=True), nullable=False, server_default=func.now())
    updated_at = Column(TIMESTAMP(timezone=True), nullable=False, server_default=func.now(), onupdate=func.now())
    
    # 关系定义
    topic = relationship("Topic", back_populates="categories")
    room_associations = relationship("TopicCategoryRoom", back_populates="category", cascade="all, delete-orphan")

    # 关系定义
    category = relationship("TopicCategory", back_populates="room_associations")

"""
LiveCore Service - Topic Schemas

专题聚合功能的Pydantic Schema定义
"""

from pydantic import BaseModel, Field, validator
from typing import Optional, List
from datetime import datetime
from enum import Enum
import uuid


class TopicStatus(str, Enum):
    """专题状态枚举"""
    DRAFT = "draft"
    PUBLISHED = "published"
    ARCHIVED = "archived"


# ===== 专题 Schema =====
class TopicBase(BaseModel):
    """专题基础Schema"""
    title: str = Field(..., min_length=1, max_length=100, description="专题标题")
    description: Optional[str] = Field(None, description="专题描述")
    banner_url: Optional[str] = Field(None, max_length=255, description="横幅图URL")
    status: TopicStatus = Field(TopicStatus.DRAFT, description="专题状态")


class TopicCreate(TopicBase):
    """创建专题请求Schema"""
    pass


class TopicUpdate(BaseModel):
    """更新专题请求Schema"""
    title: Optional[str] = Field(None, min_length=1, max_length=100)
    description: Optional[str] = None
    banner_url: Optional[str] = Field(None, max_length=255)
    status: Optional[TopicStatus] = None


class TopicInDB(TopicBase):
    """数据库中的专题Schema"""
    id: uuid.UUID
    user_id: uuid.UUID
    created_at: datetime
    updated_at: datetime
    
    class Config:
        from_attributes = True


class TopicResponse(TopicInDB):
    """专题响应Schema"""
    pass


# ===== 分类 Schema =====
class CategoryBase(BaseModel):
    """分类基础Schema"""
    name: str = Field(..., min_length=1, max_length=50, description="分类名称")
    sort_order: int = Field(0, ge=0, description="排序顺序")


class CategoryCreate(CategoryBase):
    """创建分类请求Schema"""
    pass


class CategoryUpdate(BaseModel):
    """更新分类请求Schema"""
    name: Optional[str] = Field(None, min_length=1, max_length=50)
    sort_order: Optional[int] = Field(None, ge=0)


class CategoryInDB(CategoryBase):
    """数据库中的分类Schema"""
    id: uuid.UUID
    topic_id: uuid.UUID
    created_at: datetime
    updated_at: datetime
    
    class Config:
        from_attributes = True


class CategoryResponse(CategoryInDB):
    """分类响应Schema"""
    pass


# ===== 关联管理 Schema =====
class RoomAssociation(BaseModel):
    """单个直播间关联Schema"""
    room_id: uuid.UUID
    sort_order: int = Field(0, ge=0)


class AddRoomsRequest(BaseModel):
    """添加直播间到分类请求Schema"""
    rooms: List[RoomAssociation] = Field(..., min_items=1, max_items=50)
    
    @validator('rooms')
    def validate_unique_room_ids(cls, v):
        """验证room_id唯一性"""
        room_ids = [r.room_id for r in v]
        if len(room_ids) != len(set(room_ids)):
            raise ValueError("room_id 不能重复")
        return v


class UpdateRoomSortRequest(BaseModel):
    """更新直播间排序请求Schema"""
    rooms: List[RoomAssociation] = Field(..., min_items=1, max_items=100)


class RemoveRoomsRequest(BaseModel):
    """移除直播间请求Schema"""
    room_ids: List[uuid.UUID] = Field(..., min_items=1, max_items=50)


class BatchStatusRequest(BaseModel):
    """批量查询状态请求Schema"""
    room_ids: List[uuid.UUID] = Field(..., min_items=1, max_items=100)


# ===== 聚合页响应 Schema =====
class RoomInCategory(BaseModel):
    """分类下的直播间Schema"""
    id: uuid.UUID
    title: str
    cover_url: Optional[str]
    live_status: str
    start_time: Optional[datetime]
    heat: int


class CategoryWithRooms(BaseModel):
    """包含直播间列表的分类Schema"""
    id: uuid.UUID
    name: str
    sort_order: int
    rooms: List[RoomInCategory]


class TopicDetailResponse(TopicInDB):
    """专题详情响应Schema（包含层级化数据）"""
    categories: List[CategoryWithRooms]
