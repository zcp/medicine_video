def sanitize_error(error: Exception) -> str:
    """
    脱敏异常信息，避免泄露敏感信息

    Args:
        error: 异常对象

    Returns:
        脱敏后的错误信息（只记录错误类型，不记录详细内容）
    """
    error_type = type(error).__name__
    error_str = str(error)

    # 只记录错误类型，不记录完整的错误信息
    # 如果错误信息中包含敏感信息（如数据库连接信息），只记录前50个字符
    if len(error_str) > 50:
        return f"{error_type}: {error_str[:50]}..."
    else:
        return f"{error_type}: {error_str}"
```
