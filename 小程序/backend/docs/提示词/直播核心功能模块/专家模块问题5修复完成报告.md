**位置**：`get_followed_experts_with_live_status()` 函数（第548-603行）

**修改后**：
```python
        # 构建结果
        result = []
        # 获取所有专家ID
        all_expert_ids = [sub.expert_id for sub in subscriptions]

        # 批量查询直播状态（如果有关注的专家）
        live_status_map = {}
        if all_expert_ids:
            # 查询每个专家最近的直播记录（仅查询进行中的直播）
            live_sessions_query = (
                db.query(
                    LiveSessionExpert.expert_id,
                    LiveSession.session_id,
                    LiveSession.status,
                    LiveSession.title,
                    LiveSession.start_time,
                    LiveSession.viewer_count
                )
                .join(LiveSession, LiveSessionExpert.session_id == LiveSession.id)
                .filter(
                    LiveSessionExpert.expert_id.in_(all_expert_ids),
                    LiveSession.status == 'live'  # 仅查询进行中的直播
                )
                .order_by(LiveSession.start_time.desc())
            )

            # 对于每个专家，取最近的一条直播记录
            for row in live_sessions_query.all():
                expert_id = row.expert_id
                if expert_id not in live_status_map:  # 每个专家只取最近的一条
                    live_status_map[expert_id] = {
                        "is_live": True,
                        "session_id": str(row.session_id),
                        "status": row.status,
                        "title": row.title,
                        "started_at": row.start_time.isoformat() if row.start_time else None,
                        "viewer_count": row.viewer_count
                    }

        # 构建结果
        for sub in subscriptions:
            expert = sub.expert
            expert_id = sub.expert_id
            # 获取该专家的直播状态（如果没有则返回空状态）
            live_status = live_status_map.get(expert_id)

            result.append({
                "expert_id": expert_id,
                "name": expert.name,
                "title": expert.title,
                "hospital": expert.hospital,
                "avatar_url": expert.avatar_url,
                "subscribed_at": sub.created_at,
                "live_status": live_status  # 包含直播状态（如果该专家正在直播）
            })
```

**实现逻辑**：
1. ✅ 获取所有关注专家的ID列表
2. ✅ 批量查询 `live_session_experts` 和 `live_sessions` 表
3. ✅ 只查询进行中的直播（`status == 'live'`）
4. ✅ 按直播开始时间降序排序，取每个专家最近的一条直播记录
5. ✅ 构建直播状态映射表（`live_status_map`）
6. ✅ 将直播状态组装到每个专家的返回结果中

**查询SQL**：
```sql
SELECT
    lse.expert_id,
    ls.id as session_id,
    ls.status,
    ls.title,
    ls.start_time,
    ls.viewer_count
FROM live_session_experts lse
JOIN live_sessions ls ON lse.session_id = ls.id
WHERE lse.expert_id IN (:expert_ids)
  AND ls.status = 'live'
ORDER BY ls.start_time DESC
```

async def get_followed_experts(
    page: int = 1,
    size: int = 20,
    include_live_status: Optional[bool] = True,  # ✅ 已包含该参数
    db: Session = Depends(get_db),
    current_user: dict = Depends(get_current_user)
):
    """
    获取关注的专家列表（分页，需登录）

    参数:
        page: 页码
        size: 每页数量
        include_live_status: 是否包含直播状态，默认true  # ✅ 已注释说明
    """
    # 解析用户身份（使用 user_id 而非 sub，遵循设计文档 Section 1.4.3 的明确要求）
    user_id = uuid.UUID(current_user.get("user_id"))

    try:
        expert_service = ExpertService(db=db)
        skip = (page - 1) * size

        # 根据include_live_status参数决定返回内容
        if include_live_status:  # ✅ 已实现根据参数决定返回内容
            # 包含直播状态
            result = expert_service.get_followed_experts_with_live_status(
                db=db,
                user_id=user_id
            )
            # 获取总数
            total = len(result)
        else:
            # 不包含直播状态
            subscriptions, total = expert_service.get_followed_experts_multi_and_total(
                db=db, user_id=user_id, skip=skip, limit=size
            )
            result = subscriptions

        return success_response(data={
            "items": result,
            "total": total,
            "page": page,
            "size": size
        })

    except Exception as e:
        logger.error(f"Failed to get followed experts: {sanitize_error(e)}")
