```sql
-- 表1: experts（专家信息表）
CREATE TABLE experts (
    id UUID PRIMARY KEY,
    user_id UUID UNIQUE NULL COMMENT '关联平台用户公开ID（users.public_id），可为空表示外部专家（未绑定平台账号），应用层验证',
    name VARCHAR(120) NOT NULL COMMENT '专家姓名',
    title VARCHAR(120) COMMENT '职称（如：主任医师、教授）',
    hospital VARCHAR(200) COMMENT '所属医院',
    department VARCHAR(120) COMMENT '科室',
    expertise_areas TEXT COMMENT '擅长领域，逗号分隔或JSON字符串',
    bio TEXT COMMENT '个人简介',
    avatar_url VARCHAR(512) COMMENT '头像URL',
    is_featured BOOLEAN DEFAULT false NOT NULL COMMENT '是否为首页推荐专家',
    sort_order INT DEFAULT 0 NOT NULL COMMENT '排序顺序',
    contact_info JSONB NULL COMMENT 'JSONB格式存储联系方式：{phone, email, office}',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '更新时间'
);

-- 索引
CREATE INDEX idx_experts_is_featured ON experts(is_featured, sort_order);
CREATE INDEX idx_experts_user_id ON experts(user_id);

-- 触发器
CREATE TRIGGER set_timestamp_experts BEFORE UPDATE ON experts
FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();


-- 表2: user_expert_subscriptions（用户专家订阅表）
CREATE TABLE user_expert_subscriptions (
    id UUID PRIMARY KEY,
    user_id UUID NOT NULL COMMENT '用户公开ID（users.public_id），应用层验证，JWT Token的user_id字段值',
    expert_id UUID NOT NULL REFERENCES experts(id) ON DELETE CASCADE COMMENT '专家ID，关联experts表',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '关注时间'
);

-- 索引
CREATE INDEX idx_user_expert_subscriptions_user_id ON user_expert_subscriptions(user_id);
CREATE INDEX idx_user_expert_subscriptions_expert_id ON user_expert_subscriptions(expert_id);
CREATE INDEX idx_user_expert_subscriptions_created_at ON user_expert_subscriptions(created_at DESC);

-- 约束
ALTER TABLE user_expert_subscriptions ADD CONSTRAINT uq_user_expert_subscriptions_user_expert UNIQUE(user_id, expert_id);


-- 表3: live_session_experts（直播场次专家关联表）
CREATE TABLE live_session_experts (
    id UUID PRIMARY KEY,
    session_id UUID NOT NULL REFERENCES live_sessions(id) ON DELETE CASCADE COMMENT '直播场次ID，关联live_sessions表',
    expert_id UUID NOT NULL REFERENCES experts(id) ON DELETE CASCADE COMMENT '专家ID，关联experts表',
    role VARCHAR(50) NOT NULL DEFAULT '主讲' COMMENT '专家角色：主讲、主持、嘉宾',
    sort_order INT DEFAULT 0 NOT NULL COMMENT '显示顺序（数值越大越靠前）',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '关联创建时间'
);

-- 索引
CREATE INDEX idx_live_session_experts_session ON live_session_experts(session_id);
CREATE INDEX idx_live_session_experts_expert ON live_session_experts(expert_id);
CREATE INDEX idx_live_session_experts_role ON live_session_experts(session_id, role);

-- 约束
ALTER TABLE live_session_experts ADD CONSTRAINT uq_live_session_experts_session_expert_role UNIQUE(session_id, expert_id, role);
```

## 5. 审查标准 (Audit Standards)

审查时请遵循以下标准：

### 5.1. 严格标准 (Strict Standards)

以下项目必须100%符合DDL规范，任何偏差都算作**失败**:
- 表名、字段名
- 数据类型（特别是 TIMESTAMP vs DateTime）
- 外键约束和级联规则
- NOT NULL 约束
- UNIQUE 约束

### 5.2. 宽松标准 (Flexible Standards)

以下项目允许合理的实现差异：
- 索引名称（只要能清晰表达含义即可）
- 注释内容（只要能说明用途即可）
- 关系名称（只要符合Python命名规范即可）
- `__repr__` 方法（可选）

### 5.3. 推荐实践 (Best Practices)

以下项目不影响功能，但建议遵循：
- 添加 `__repr__` 方法便于调试
- 使用 `lazy="selectin"` 避免 N+1 查询
- 添加详细的中文注释
- 使用 `comment` 参数为字段添加数据库注释

1. 将 `backend/live_core_service/app/models/experts.py` 的完整内容粘贴到 Section 3.2
2. 运行此提示词，让 AI 生成详细的审查报告
3. 根据报告中的"关键问题清单"和"修正代码示例"进行修复
4. 重新运行审查，直到所有检查项通过

### 6.2. 审查重点

特别关注以下容易出错的地方：
1. ⚠️ **时间戳字段** - 必须使用 `TIMESTAMP` + `server_default=func.now()`
2. ⚠️ **索引定义** - 必须使用 `__table_args__` 显式命名
3. ⚠️ **枚举类型** - 必须使用 `SessionExpertRole` 枚举
4. ⚠️ **外键级联** - `ondelete="CASCADE"` 大写
5. ⚠️ **JSONB字段** - 必须使用 `JSONB` 类型
