# 爬虫与批量导入集成方案

**文档版本**: v1.0  
**创建日期**: 2025-01-XX  
**问题分析**: 如何将爬虫程序与媒体下载服务的批量导入功能集成，并解决token管理问题

---

## 一、现状分析

### 1.1 现有组件

1. **爬虫程序** (`backend/dynamicCrawler_vzan.py`):
   - 功能：从指定URL爬取直播间数据并保存为CSV
   - 问题：token硬编码在代码中
   - 输出：CSV文件（直播间列表）

2. **批量导入接口** (`POST /api/v1/download/tasks/batch-import`):
   - 功能：接受上传的CSV文件并批量创建下载任务
   - 输入：CSV文件（multipart/form-data）
   - 认证：需要JWT Token
   - 去重：基于`resource_url + liveroom_id + liveroom_title`组合

### 1.2 存在的问题

| 问题 | 描述 | 影响 |
|:---|:---|:---|
| **问题1** | 爬虫和导入没有集成 | 需要手动操作两个步骤 |
| **问题2** | vzan token硬编码 | 安全性差，难以管理和更新 |
| **问题3** | 没有自动化流程 | 无法定时自动执行 |
| **问题4** | 错误处理不完善 | 爬取或导入失败时缺少通知 |

---

## 二、解决方案对比

### 2.1 方案对比表

| 方案 | 优点 | 缺点 | 推荐度 |
|:---|:---|:---|:---:|
| **方案1：API调用集成** | ✓ 解耦合<br>✓ 无需修改现有代码<br>✓ 灵活性高 | △ 需要处理HTTP调用<br>△ 需要管理JWT Token | ⭐⭐⭐⭐⭐ |
| **方案2：文件监控集成** | ✓ 简单<br>✓ 自动化 | ✗ 需要新增接口<br>✗ 文件同步问题 | ⭐⭐⭐ |
| **方案3：服务内集成** | ✓ 统一管理 | ✗ 代码耦合<br>✗ 维护复杂 | ⭐⭐ |

### 2.2 推荐方案：API调用集成

**核心思路**：
1. 爬虫程序生成CSV文件后，通过HTTP请求调用批量导入API
2. 使用配置文件管理所有token和配置项
3. 提供完整的错误处理和日志记录

```python
"""
媒体下载服务批量导入客户端
负责调用批量导入API，将CSV文件导入到媒体下载服务
"""
import requests
import logging
from pathlib import Path
from typing import Dict, Optional
from dynamicCrawler_vzan_config import config

logger = logging.getLogger(__name__)


class MediaDownloadImportClient:
    """媒体下载服务批量导入客户端"""
    
    def __init__(self, jwt_token: Optional[str] = None, base_url: Optional[str] = None):
        """
        初始化客户端
        
        Args:
            jwt_token: JWT认证Token（如不提供则从配置读取）
            base_url: API基础URL（如不提供则从配置读取）
        """
        self.jwt_token = jwt_token or config.MEDIA_DOWNLOAD_JWT_TOKEN
        self.base_url = base_url or config.MEDIA_DOWNLOAD_API_BASE_URL
        
        if not self.jwt_token:
            raise ValueError("JWT Token未配置，无法调用媒体下载服务API")
        
        self.session = requests.Session()
        self.session.headers.update({
            "Authorization": f"Bearer {self.jwt_token}"
        })
        
        logger.info(f"初始化批量导入客户端 - API地址: {self.base_url}")
    
    def import_csv_file(
        self,
        csv_file_path: Path,
        skip_duplicates: bool = True,
        auto_start: bool = False
    ) -> Dict:
        """
        导入CSV文件到媒体下载服务
        
        Args:
            csv_file_path: CSV文件路径
            skip_duplicates: 是否跳过重复任务（基于resource_url + liveroom_id + liveroom_title）
            auto_start: 导入后是否自动启动任务
        
        Returns:
            导入结果字典，包含total, success, failed, skipped等字段
        
        Raises:
            FileNotFoundError: CSV文件不存在
            requests.RequestException: API请求失败
            ValueError: API返回错误
        """
        # 1. 验证文件存在
        if not csv_file_path.exists():
            raise FileNotFoundError(f"CSV文件不存在: {csv_file_path}")
        
        logger.info(f"开始导入CSV文件: {csv_file_path}")
        logger.info(f"导入参数 - 跳过重复: {skip_duplicates}, 自动启动: {auto_start}")
        
        try:
            # 2. 准备multipart/form-data请求
            with open(csv_file_path, 'rb') as f:
                files = {
                    'file': (csv_file_path.name, f, 'text/csv')
                }
                data = {
                    'skip_duplicates': str(skip_duplicates).lower(),
                    'auto_start': str(auto_start).lower()
                }
                
                # 3. 调用批量导入API
                url = f"{self.base_url}/tasks/batch-import"
                logger.debug(f"请求URL: {url}")
                logger.debug(f"请求数据: {data}")
                
                response = self.session.post(
                    url,
                    files=files,
                    data=data,
                    timeout=60  # 批量导入可能耗时较长
                )
            
            # 4. 处理响应
            logger.info(f"API响应状态码: {response.status_code}")
            
            # 解析JSON响应
            response_json = response.json()
            logger.debug(f"API响应内容: {response_json}")
            
            # 检查业务状态码（在body的code字段）
            business_code = response_json.get("code")
            message = response_json.get("message", "")
            data = response_json.get("data", {})
            
            # 5. 根据业务状态码判断结果
            if business_code == 201:
                # 全部成功
                logger.info(f"✓ 批量导入成功 - {message}")
                logger.info(
                    f"  总数: {data.get('total')}, "
                    f"成功: {data.get('success')}, "
                    f"失败: {data.get('failed')}, "
                    f"跳过: {data.get('skipped')}"
                )
                return data
            
            elif business_code == 207:
                # 部分成功
                logger.warning(f"△ 批量导入部分成功 - {message}")
                logger.warning(
                    f"  总数: {data.get('total')}, "
                    f"成功: {data.get('success')}, "
                    f"失败: {data.get('failed')}, "
                    f"跳过: {data.get('skipped')}"
                )
                
                # 记录失败行详情
                failed_rows = data.get('failed_rows', [])
                if failed_rows:
                    logger.warning(f"  失败行数: {len(failed_rows)}")
                    for row in failed_rows[:5]:  # 只显示前5行
                        logger.warning(
                            f"    行{row.get('row')}: {row.get('error')}"
                        )
                
                # 记录跳过行详情
                skipped_rows = data.get('skipped_rows', [])
                if skipped_rows:
                    logger.info(f"  跳过行数: {len(skipped_rows)} (重复任务)")
                
                return data
            
            else:
                # 失败
                error_msg = f"批量导入失败 - {message}"
                logger.error(f"✗ {error_msg}")
                raise ValueError(error_msg)
        
        except requests.RequestException as e:
            error_msg = f"API请求失败: {str(e)}"
            logger.error(error_msg, exc_info=True)
            raise
        
        except Exception as e:
            error_msg = f"导入过程发生错误: {str(e)}"
            logger.error(error_msg, exc_info=True)
            raise
    
    def test_connection(self) -> bool:
        """
        测试与媒体下载服务的连接
        
        Returns:
            连接是否成功
        """
        try:
            # 尝试获取任务列表（只取1条）
            url = f"{self.base_url}/tasks?page=1&size=1"
            response = self.session.get(url, timeout=10)
            
            if response.status_code == 200:
                logger.info("✓ 媒体下载服务连接测试成功")
                return True
            else:
                logger.warning(f"△ 媒体下载服务连接测试失败 - 状态码: {response.status_code}")
                return False
        
        except Exception as e:
            logger.error(f"✗ 媒体下载服务连接测试失败: {str(e)}")
            return False


# 便捷函数
def import_csv_to_media_download(
    csv_file_path: Path,
    skip_duplicates: bool = None,
    auto_start: bool = None,
    jwt_token: str = None
) -> Dict:
    """
    便捷函数：导入CSV文件到媒体下载服务
    
    Args:
        csv_file_path: CSV文件路径
        skip_duplicates: 是否跳过重复任务（如不提供则从配置读取）
        auto_start: 导入后是否自动启动任务（如不提供则从配置读取）
        jwt_token: JWT Token（如不提供则从配置读取）
    
    Returns:
        导入结果
    """
    # 使用配置的默认值
    if skip_duplicates is None:
        skip_duplicates = config.SKIP_DUPLICATES_ON_IMPORT
    
    if auto_start is None:
        auto_start = config.AUTO_START_AFTER_IMPORT
    
    # 创建客户端并导入
    client = MediaDownloadImportClient(jwt_token=jwt_token)
    return client.import_csv_file(csv_file_path, skip_duplicates, auto_start)
```

### 5.1 Token更新策略

#### 5.1.1 VZAN Token更新

**问题**: VZAN Token会过期，需要定期更新

**解决方案**:

```python
# 在dynamicCrawler_vzan_config.py中添加

import jwt
from datetime import datetime, timedelta

class TokenManager:
    """Token管理器"""
    
    @staticmethod
    def check_token_expiry(token: str) -> dict:
        """
        检查Token过期时间
        
        Returns:
            {
                'valid': bool,
                'expires_at': datetime,
                'days_until_expiry': int
            }
        """
        try:
            # 解码JWT（不验证签名，仅获取payload）
            payload = jwt.decode(token, options={"verify_signature": False})
            
            exp_timestamp = payload.get('exp')
            if not exp_timestamp:
                return {'valid': False, 'reason': 'No expiration'}
            
            expires_at = datetime.fromtimestamp(exp_timestamp)
            now = datetime.now()
            days_until_expiry = (expires_at - now).days
            
            return {
                'valid': expires_at > now,
                'expires_at': expires_at,
                'days_until_expiry': days_until_expiry
            }
        
        except Exception as e:
            return {'valid': False, 'reason': str(e)}
    
    @staticmethod
    def should_renew_token(token: str, threshold_days: int = 7) -> bool:
        """判断是否需要续期Token"""
        info = TokenManager.check_token_expiry(token)
        if not info.get('valid'):
            return True
        return info.get('days_until_expiry', 0) < threshold_days


# 在爬虫初始化时检查
def check_tokens_before_crawl():
    """爬取前检查Token有效性"""
    logger = logging.getLogger(__name__)
    
    # 检查VZAN Token
    vzan_token_info = TokenManager.check_token_expiry(config.VZAN_TOKEN)
    if not vzan_token_info.get('valid'):
        logger.error(f"✗ VZAN Token无效或已过期: {vzan_token_info.get('reason')}")
        raise ValueError("VZAN Token无效，请更新")
    
    days_left = vzan_token_info.get('days_until_expiry')
    if days_left < 7:
        logger.warning(f"△ VZAN Token将在{days_left}天后过期，请及时更新")
    else:
        logger.info(f"✓ VZAN Token有效，剩余{days_left}天")
    
    # 检查媒体下载JWT Token
    jwt_token_info = TokenManager.check_token_expiry(config.MEDIA_DOWNLOAD_JWT_TOKEN)
    if not jwt_token_info.get('valid'):
        logger.error(f"✗ 媒体下载JWT Token无效或已过期")
        if config.AUTO_IMPORT_AFTER_CRAWL:
            raise ValueError("JWT Token无效，无法自动导入")
    else:
        logger.info(f"✓ 媒体下载JWT Token有效")
```

from cryptography.fernet import Fernet
import os

class SecureTokenManager:
    """加密Token管理"""
    
    def __init__(self):
        # 从环境变量读取加密密钥
        key = os.getenv("ENCRYPTION_KEY")
        if not key:
            raise ValueError("ENCRYPTION_KEY未配置")

        """加密Token"""
        return self.cipher.encrypt(token.encode()).decode()
    
    def decrypt_token(self, encrypted_token: str) -> str:
        """解密Token"""
        return self.cipher.decrypt(encrypted_token.encode()).decode()
